PATCH codi_x86_64U.finc  --  palier 3 : entree console (SYS_GET_CHAR, SYS_GET_STR)
Base : le fichier issu du palier 2.
Verifie sous OVMF : GET_CHAR et GET_LINE fonctionnent, y compris le retour arriere,
la ligne vide, le depassement de tampon, et le CR ignore par GET au sens LRM.

Trois retouches.
  1. .data : les deux sequences d'echo (retour arriere destructif, CR LF)
  2. .bss  : tampon d'echo, EFI_INPUT_KEY, evenement et index
  3. le bloc d'entree : SYS_GET_CHAR et SYS_GET_STR reecrits, precedes des deux
     briques UEFI_CONOUT et UEFI_READ_KEY

========================================================================================================
PATCH 1  --  .data : sequences d'echo, en UTF-16
========================================================================================================
--- ANCRE (ne pas modifier, sert a se reperer) --------------------------------------------------------
UEFI_MARK_EXIT:(inserer les deux lignes ci-dessous JUSTE AVANT l'ancre)
--- BLOC A INSERER -----------------------------------------------------------------------------------
UEFI_BS_SEQ:	db 8,0,32,0,8,0,0,0				; retour arriere destructif (BS, espace, BS)
UEFI_CRLF_SEQ:	db 13,0,10,0,0,0

========================================================================================================
PATCH 2  --  .bss : etat de l'entree clavier
========================================================================================================
--- ANCRE (ne pas modifier, sert a se reperer) --------------------------------------------------------
UEFI_TXT_LAST:		rb 1					; dernier octet emis (pour ne pas doubler le CR)
--- BLOC A INSERER -----------------------------------------------------------------------------------
UEFI_ECHO_BUF:		rw 4					; un CHAR16 + terminateur, pour l'echo de SYS_GET_STR
UEFI_KEY:		rw 2					; EFI_INPUT_KEY { ScanCode, UnicodeChar }
UEFI_KEY_EVENT:		rq 1					; copie de ConIn->WaitForKey pour WaitForEvent
UEFI_KEY_INDEX:		rq 1					; index rendu par WaitForEvent (inutilise)

========================================================================================================
PATCH 3  --  le bloc d'entree console
========================================================================================================
Tout le bloc, depuis la ligne de commentaire qui precede SYS_GET_CHAR jusqu'au
"end macro" qui clot SYS_GET_STR, est remplace. Les deux macros gardent exactement
leur contrat de pile : SYS_GET_CHAR depile @ITEM, SYS_GET_STR depile @descripteur
puis @LAST. TEXT_IO n'est pas touche.
--- BLOC A SUPPRIMER ---------------------------------------------------------------------------------
;                       ------------
macro                   SYS_GET_CHAR
;                       ------------
  local no_console, get_char_done

  POP_RSI                                             ; RSI = @ITEM (destination)

  ; Cadre Win64 local :
  ;   rsp+00..31 : shadow space
  ;   rsp+32     : 5e argument ReadFile = lpOverlapped = NULL
  ;   rsp+40     : DWORD bytes_read
  ;   rsp+44     : DWORD old_console_mode
  ;   rsp+64     : ancien RSP TLALOC
  db 0x48, 0x89, 0xE0                                ; mov rax,rsp
  db 0x48, 0x83, 0xE4, 0xF0                          ; and rsp,-16
  db 0x48, 0x83, 0xEC, 0x50                          ; sub rsp,80
  db 0x48, 0x89, 0x44, 0x24, 0x40                    ; mov [rsp+64],rax
  db 0x48, 0xC7, 0x44, 0x24, 0x20, 0, 0, 0, 0       ; [rsp+32] = NULL
  db 0xC7, 0x44, 0x24, 0x28, 0, 0, 0, 0             ; [rsp+40] = bytes_read = 0

  ; GetConsoleMode( stdin, &old_mode )
  db 0x48, 0x31, 0xC9						; xor rcx,rcx : handle absent sous UEFI
  db 0x48, 0x8D, 0x54, 0x24, 0x2C                    ; lea rdx,[rsp+44]
  db 0x31, 0xC0							; xor eax,eax : service non porte -> branche d'echec
  db 0x85, 0xC0                                      ; test eax,eax
  db 0x0F, 0x84                                      ; jz no_console
  dd no_console - ($ + 4)

  ; SetConsoleMode( stdin, old_mode & ~(ENABLE_LINE_INPUT|ENABLE_ECHO_INPUT) )
  db 0x8B, 0x54, 0x24, 0x2C                          ; mov edx,[rsp+44]
  db 0x83, 0xE2, 0xF9                                ; and edx,FFFFFFF9h
  db 0x48, 0x31, 0xC9						; xor rcx,rcx : handle absent sous UEFI
  db 0x31, 0xC0							; xor eax,eax : service non porte -> branche d'echec

  ; ReadFile( stdin, @ITEM, 1, &bytes_read, NULL )
  db 0x48, 0x31, 0xC9						; xor rcx,rcx : handle absent sous UEFI
  db 0x48, 0x89, 0xF2                                ; mov rdx,rsi
  db 0x41, 0xB8, 1, 0, 0, 0                          ; mov r8d,1
  db 0x4C, 0x8D, 0x4C, 0x24, 0x28                    ; lea r9,[rsp+40]
  db 0x31, 0xC0							; xor eax,eax : service non porte -> branche d'echec

  ; Restaurer exactement le mode d'origine.
  db 0x8B, 0x54, 0x24, 0x2C                          ; mov edx,[rsp+44]
  db 0x48, 0x31, 0xC9						; xor rcx,rcx : handle absent sous UEFI
  db 0x31, 0xC0							; xor eax,eax : service non porte -> branche d'echec
  db 0xE9                                             ; jmp get_char_done
  dd get_char_done - ($ + 4)

no_console:
  ; stdin redirige (fichier/pipe) : GetConsoleMode est inapplicable,
  ; mais ReadFile sur le handle standard reste valide.
  db 0x48, 0x31, 0xC9						; xor rcx,rcx : handle absent sous UEFI
  db 0x48, 0x89, 0xF2                                ; mov rdx,rsi
  db 0x41, 0xB8, 1, 0, 0, 0                          ; mov r8d,1
  db 0x4C, 0x8D, 0x4C, 0x24, 0x28                    ; lea r9,[rsp+40]
  db 0x31, 0xC0							; xor eax,eax : service non porte -> branche d'echec

get_char_done:
  db 0x48, 0x8B, 0x64, 0x24, 0x40                    ; mov rsp,[rsp+64]
end macro


; SYS_GET_STR : lit une ligne en mode console normal (ligne + echo).
; A l'entree LLIR : @LAST puis @descripteur STRING ont ete empiles ; le
; descripteur est donc depile en premier. ReadFile est volontairement utilise
; plutot que ReadConsole afin que stdin redirige continue de fonctionner.
; On retranche du compte rendu un LF et/ou CR final, sans enlever un vrai
; caractere lorsque le buffer est rempli avant le terminateur de ligne.
;
;                       -----------
macro                   SYS_GET_STR
;                       -----------
  local check_cr, store_last

  POP_RSI                                             ; RSI = @(@DATA | @USEINFO)
  db 0x48, 0x8B, 0x46, 0x08                          ; mov rax,[rsi+8] : @USEINFO
  db 0x44, 0x8B, 0x40, 0x0C                          ; mov r8d,[rax+12] : last
  db 0x41, 0xFF, 0xC0                                ; inc r8d
  db 0x44, 0x2B, 0x40, 0x08                          ; sub r8d,[rax+8] : buffer length
  db 0x48, 0x8B, 0x16                                ; mov rdx,[rsi] : @DATA
  POP_RSI                                             ; RSI = @LAST (resultat Ada)

  ; Cadre Win64 : shadow, lpOverlapped, bytes_read, ancien RSP, @DATA.
  db 0x48, 0x89, 0xE0                                ; mov rax,rsp
  db 0x48, 0x83, 0xE4, 0xF0                          ; and rsp,-16
  db 0x48, 0x83, 0xEC, 0x40                          ; sub rsp,64
  db 0x48, 0x89, 0x44, 0x24, 0x30                    ; mov [rsp+48],rax
  db 0x48, 0x89, 0x54, 0x24, 0x38                    ; mov [rsp+56],rdx : sauver @DATA
  db 0x48, 0xC7, 0x44, 0x24, 0x20, 0, 0, 0, 0       ; [rsp+32] = NULL
  db 0xC7, 0x44, 0x24, 0x28, 0, 0, 0, 0             ; [rsp+40] = bytes_read = 0

  ; ReadFile( stdin, @DATA, length, &bytes_read, NULL )
  db 0x48, 0x31, 0xC9						; xor rcx,rcx : handle absent sous UEFI
  db 0x4C, 0x8D, 0x4C, 0x24, 0x28                    ; lea r9,[rsp+40]
  db 0x31, 0xC0							; xor eax,eax : service non porte -> branche d'echec

  ; EAX = nombre d'octets recus. Retirer seulement les terminateurs presents.
  db 0x8B, 0x44, 0x24, 0x28                          ; mov eax,[rsp+40]
  db 0x48, 0x8B, 0x54, 0x24, 0x38                    ; mov rdx,[rsp+56] : @DATA
  db 0x85, 0xC0                                      ; test eax,eax
  db 0x0F, 0x84                                      ; jz store_last
  dd store_last - ($ + 4)
  db 0x80, 0x7C, 0x02, 0xFF, 0x0A                    ; cmp byte [rdx+rax-1],LF
  db 0x0F, 0x85                                      ; jne check_cr
  dd check_cr - ($ + 4)
  db 0xFF, 0xC8                                      ; dec eax

check_cr:
  db 0x85, 0xC0                                      ; test eax,eax
  db 0x0F, 0x84                                      ; jz store_last
  dd store_last - ($ + 4)
  db 0x80, 0x7C, 0x02, 0xFF, 0x0D                    ; cmp byte [rdx+rax-1],CR
  db 0x0F, 0x85                                      ; jne store_last
  dd store_last - ($ + 4)
  db 0xFF, 0xC8                                      ; dec eax

store_last:
  db 0x89, 0x06                                      ; mov [rsi],eax : LAST/longueur utile
  db 0x48, 0x8B, 0x64, 0x24, 0x30                    ; mov rsp,[rsp+48]
end macro
--- BLOC A INSERER -----------------------------------------------------------------------------------
;-----------------------------------------------------------------------------------------------------------------------
;			E N T R E E   C O N S O L E   U E F I
;-----------------------------------------------------------------------------------------------------------------------
;	Windows livrait une console avec discipline de ligne : ReadFile rendait une
;	ligne entiere, deja affichee, deja corrigible au retour arriere, terminee par
;	CR LF -- et GetConsoleMode/SetConsoleMode permettaient de debrayer tout cela
;	pour lire un caractere brut.
;
;	UEFI ne fournit rien de tel : ConIn->ReadKeyStroke rend UNE frappe, sans echo,
;	sans edition, sans attente. Le partage des roles s'inverse donc :
;	  - SYS_GET_CHAR devient le cas SIMPLE (c'est exactement ReadKeyStroke) ;
;	  - SYS_GET_STR doit reconstituer la discipline de ligne a la main : echo,
;	    retour arriere, arret sur CR, garde de debordement.
;
;	ReadKeyStroke rend EFI_NOT_READY si rien n'est en attente : on se met donc en
;	attente sur l'evenement ConIn->WaitForKey via BootServices->WaitForEvent, au
;	lieu de scruter en boucle et de saturer le processeur.
;
;	EFI_INPUT_KEY = { UINT16 ScanCode ; CHAR16 UnicodeChar }.  UnicodeChar vaut 0
;	pour les touches sans caractere (fleches, fonctions) : on les ignore.

ConIn			= 0x30				; dans EFI_SYSTEM_TABLE
ReadKeyStroke		= 0x08				; dans EFI_SIMPLE_TEXT_INPUT_PROTOCOL
WaitForKey		= 0x10				;   idem (EFI_EVENT, pas une fonction)
WaitForEvent		= 0x60				; dans EFI_BOOT_SERVICES

;			-----------
macro			UEFI_CONOUT					; ConOut->OutputString( ConOut, RDX )
;			-----------
;	RDX contient deja le CHAR16* a afficher. Ne touche que des registres volatils
;	de l'ABI Microsoft x64 : RBX, RSI et RDI survivent, ce dont la boucle de
;	SYS_GET_STR a besoin.
  db 0x4C, 0x8B, 0x15						; mov r10,[rip+UEFI_SYSTEM_TABLE]
  dd UEFI_SYSTEM_TABLE - ($ + 4)
  db 0x49, 0x8B, 0x4A, ConOut					; mov rcx,[r10+0x40]	RCX = ConOut
  db 0x4C, 0x8B, 0x51, OutputString				; mov r10,[rcx+0x08]	R10 = &OutputString
  db 0x48, 0x89, 0xE0						; mov rax,rsp
  db 0x48, 0x83, 0xE4, 0xF0					; and rsp,-16
  db 0x48, 0x83, 0xEC, 0x30					; sub rsp,48
  db 0x48, 0x89, 0x44, 0x24, 0x28				; mov [rsp+40],rax
  db 0x41, 0xFF, 0xD2						; call r10
  db 0x48, 0x8B, 0x64, 0x24, 0x28				; mov rsp,[rsp+40]
end macro

;			-------------
macro			UEFI_READ_KEY					; attend une frappe -> EAX = UnicodeChar
;			-------------
  local k_retry
k_retry:
			; 1. BootServices->WaitForEvent( 1, &ConIn->WaitForKey, &index )
  db 0x4C, 0x8B, 0x15						; mov r10,[rip+UEFI_SYSTEM_TABLE]
  dd UEFI_SYSTEM_TABLE - ($ + 4)
  db 0x49, 0x8B, 0x4A, ConIn					; mov rcx,[r10+0x30]	RCX = ConIn
  db 0x48, 0x8B, 0x41, WaitForKey				; mov rax,[rcx+0x10]	l'evenement
  db 0x48, 0x89, 0x05						; mov [rip+UEFI_KEY_EVENT],rax
  dd UEFI_KEY_EVENT - ($ + 4)
  db 0x4D, 0x8B, 0x52, BootServices				; mov r10,[r10+0x60]
  db 0x4D, 0x8B, 0x52, WaitForEvent				; mov r10,[r10+0x60]	&WaitForEvent
  db 0xB9, 1, 0, 0, 0						; mov ecx,1		un seul evenement
  db 0x48, 0x8D, 0x15						; lea rdx,[rip+UEFI_KEY_EVENT]
  dd UEFI_KEY_EVENT - ($ + 4)
  db 0x4C, 0x8D, 0x05						; lea r8,[rip+UEFI_KEY_INDEX]
  dd UEFI_KEY_INDEX - ($ + 4)
  db 0x48, 0x89, 0xE0						; mov rax,rsp
  db 0x48, 0x83, 0xE4, 0xF0					; and rsp,-16
  db 0x48, 0x83, 0xEC, 0x30					; sub rsp,48
  db 0x48, 0x89, 0x44, 0x24, 0x28				; mov [rsp+40],rax
  db 0x41, 0xFF, 0xD2						; call r10
  db 0x48, 0x8B, 0x64, 0x24, 0x28				; mov rsp,[rsp+40]
			; 2. ConIn->ReadKeyStroke( ConIn, &UEFI_KEY )
  db 0x4C, 0x8B, 0x15						; mov r10,[rip+UEFI_SYSTEM_TABLE]
  dd UEFI_SYSTEM_TABLE - ($ + 4)
  db 0x49, 0x8B, 0x4A, ConIn					; mov rcx,[r10+0x30]	RCX = ConIn
  db 0x4C, 0x8B, 0x51, ReadKeyStroke				; mov r10,[rcx+0x08]	R10 = &ReadKeyStroke
  db 0x48, 0x8D, 0x15						; lea rdx,[rip+UEFI_KEY]
  dd UEFI_KEY - ($ + 4)
  db 0x48, 0x89, 0xE0						; mov rax,rsp
  db 0x48, 0x83, 0xE4, 0xF0					; and rsp,-16
  db 0x48, 0x83, 0xEC, 0x30					; sub rsp,48
  db 0x48, 0x89, 0x44, 0x24, 0x28				; mov [rsp+40],rax
  db 0x41, 0xFF, 0xD2						; call r10
  db 0x48, 0x8B, 0x64, 0x24, 0x28				; mov rsp,[rsp+40]
  db 0x48, 0x85, 0xC0						; test rax,rax		EFI_SUCCESS ?
  db 0x0F, 0x85							; jnz k_retry		(course perdue : on reattend)
  dd k_retry - ($ + 4)
  db 0x0F, 0xB7, 0x05						; movzx eax,word [rip+UEFI_KEY+2]  UnicodeChar
  dd UEFI_KEY + 2 - ($ + 4)
end macro

;                       ------------
macro                   SYS_GET_CHAR					; stdin UEFI, sans echo
;                       ------------
;	Contrat inchange : depile @ITEM, y range un octet. Pas d'echo, comme la
;	version Windows qui coupait ENABLE_ECHO_INPUT le temps de la lecture.
  local c_loop
  POP_RSI								; RSI = @ITEM (destination)
c_loop:
  UEFI_READ_KEY
  db 0x66, 0x85, 0xC0						; test ax,ax
  db 0x0F, 0x84
  dd c_loop - ($ + 4)					; jz c_loop		touche sans caractere
  db 0x88, 0x06							; mov [rsi],al
end macro

;                       -----------
macro                   SYS_GET_STR					; stdin UEFI, discipline de ligne a la main
;                       -----------
;	Contrat inchange : depile @descripteur puis @LAST, remplit DATA et range la
;	longueur utile dans LAST. Le terminateur n'est jamais stocke, donc le
;	rognage du CR LF que faisait la version Windows n'a plus lieu d'etre.
;
;	Cadre local (RSP aligne 16, les appels firmware creent le leur en dessous) :
;	  rsp+00..31 : shadow space des appels imbriques
;	  rsp+32     : @LAST
;	  rsp+40     : longueur maximale du tampon Ada (dword)
;	  rsp+48     : ancien RSP TLALOC
;	Etat de boucle dans RBX (compte) et RDI (ecriture) : non volatils.
  local g_loop, g_bs, g_eol
  POP_RSI								; RSI = @(@DATA | @USEINFO)
  db 0x48, 0x8B, 0x46, 0x08					; mov rax,[rsi+8] : @USEINFO
  db 0x44, 0x8B, 0x40, 0x0C					; mov r8d,[rax+12] : last
  db 0x41, 0xFF, 0xC0						; inc r8d
  db 0x44, 0x2B, 0x40, 0x08					; sub r8d,[rax+8] : longueur du tampon
  db 0x48, 0x8B, 0x3E						; mov rdi,[rsi] : @DATA
  POP_RSI								; RSI = @LAST (resultat Ada)

  db 0x48, 0x89, 0xE0						; mov rax,rsp
  db 0x48, 0x83, 0xE4, 0xF0					; and rsp,-16
  db 0x48, 0x83, 0xEC, 0x40					; sub rsp,64
  db 0x48, 0x89, 0x44, 0x24, 0x30				; mov [rsp+48],rax : ancien RSP
  db 0x48, 0x89, 0x74, 0x24, 0x20				; mov [rsp+32],rsi : @LAST
  db 0x44, 0x89, 0x44, 0x24, 0x28				; mov [rsp+40],r8d : longueur max
  db 0x31, 0xDB							; xor ebx,ebx : compte = 0
g_loop:
  UEFI_READ_KEY
  db 0x66, 0x85, 0xC0						; test ax,ax
  db 0x0F, 0x84
  dd g_loop - ($ + 4)					; jz g_loop		touche sans caractere
  db 0x66, 0x83, 0xF8, 0x0D					; cmp ax,13		CR = fin de ligne
  db 0x0F, 0x84
  dd g_eol - ($ + 4)					; je g_eol
  db 0x66, 0x83, 0xF8, 0x08					; cmp ax,8		BS = retour arriere
  db 0x0F, 0x84
  dd g_bs - ($ + 4)					; je g_bs
  db 0x3B, 0x5C, 0x24, 0x28					; cmp ebx,[rsp+40]
  db 0x0F, 0x83
  dd g_loop - ($ + 4)					; jae g_loop		tampon plein : frappe ignoree
  db 0x66, 0x89, 0x05						; mov [rip+UEFI_ECHO_BUF],ax	(avant de perdre AX)
  dd UEFI_ECHO_BUF - ($ + 4)
  db 0x66, 0xC7, 0x05						; mov word [rip+UEFI_ECHO_BUF+2],0
  dd UEFI_ECHO_BUF + 2 - ($ + 6)
  db 0x00, 0x00
  db 0x88, 0x05							; mov [rip+UEFI_TXT_LAST],al	(cf. UEFI_WRITE_STDOUT)
  dd UEFI_TXT_LAST - ($ + 4)
  db 0x88, 0x07							; mov [rdi],al
  db 0x48, 0xFF, 0xC7						; inc rdi
  db 0xFF, 0xC3							; inc ebx
  db 0x48, 0x8D, 0x15						; lea rdx,[rip+UEFI_ECHO_BUF]
  dd UEFI_ECHO_BUF - ($ + 4)
  UEFI_CONOUT								; echo : UEFI n'en fait aucun
  db 0xE9
  dd g_loop - ($ + 4)					; jmp g_loop
g_bs:
  db 0x85, 0xDB							; test ebx,ebx
  db 0x0F, 0x84
  dd g_loop - ($ + 4)					; jz g_loop		rien a effacer
  db 0x48, 0xFF, 0xCF						; dec rdi
  db 0xFF, 0xCB							; dec ebx
  db 0x48, 0x8D, 0x15						; lea rdx,[rip+UEFI_BS_SEQ]
  dd UEFI_BS_SEQ - ($ + 4)
  UEFI_CONOUT								; BS, espace, BS : effacer a l'ecran
  db 0xC6, 0x05							; mov byte [rip+UEFI_TXT_LAST],8
  dd UEFI_TXT_LAST - ($ + 5)
  db 0x08
  db 0xE9
  dd g_loop - ($ + 4)					; jmp g_loop
g_eol:
  db 0x48, 0x8D, 0x15						; lea rdx,[rip+UEFI_CRLF_SEQ]
  dd UEFI_CRLF_SEQ - ($ + 4)
  UEFI_CONOUT								; passer a la ligne, comme le faisait la console Windows
  db 0xC6, 0x05							; mov byte [rip+UEFI_TXT_LAST],10
  dd UEFI_TXT_LAST - ($ + 5)
  db 0x0A
  db 0x48, 0x8B, 0x74, 0x24, 0x20				; mov rsi,[rsp+32] : @LAST
  db 0x89, 0x1E							; mov [rsi],ebx : longueur utile
  db 0x48, 0x8B, 0x64, 0x24, 0x30				; mov rsp,[rsp+48]
end macro

========================================================================================================
POURQUOI CES SAUTS SONT EN FORME LONGUE
========================================================================================================
UEFI_READ_KEY fait environ 135 octets (deux appels firmware avec leur cadre).
Les boucles de SYS_GET_CHAR et SYS_GET_STR l'enjambent, donc aucun deplacement
ne tient sur un octet signe. Tous les sauts de ces deux macros sont en rel32 :
  jz  -> 0F 84 + dd    jne -> 0F 85 + dd    jae -> 0F 83 + dd    jmp -> E9 + dd
fasmg signale le probleme par "value out of allowed range" si l'on oublie.

========================================================================================================
CONTROLES
========================================================================================================
  qemu-system-x86_64 -bios /usr/share/ovmf/OVMF.fd -drive format=raw,file=fat:rw:esp -nographic

  GET_CHAR : Tape un caractere :        (rien ne s'affiche a la frappe)
             Lu = A
             Entree tapee avant le caractere : ignoree (CR est un terminateur LRM)

  GET_LINE : Ligne : Bonjour le monde   (frappe avec retours arriere)
             Recu : Bonjour le monde
             ligne vide      -> Recu :
             plus de 64 car. -> tronque a 64, frappes en trop non affichees
