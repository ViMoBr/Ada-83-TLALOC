  use IR;												--| operateurs ELT_KIND / OPERAND_TAG (LRM 8.4)

  --|  Transcription MOT DE 32 BITS PAR MOT de codi_riscv64.finc v2 (28-29 aout 2026),
  --|  la table entiere en une livraison : les sequences fixes sont transcrites
  --|  MECANIQUEMENT (gen_ada.py : lecture du codi, expansion des sous-macros,
  --|  resolution des branchements B-type, desassemblage capstone de controle) ;
  --|  les macros parametrees (QUAD_CONST, acces [a0+disp], LINK, RTD, EXC_*, LEXCMP)
  --|  sont ecrites a la main sur le modele d'ARM64_TARGET et confrontees a
  --|  l'emulateur du codi par un programme gnat (tst_rv.adb).
  --|  Contrat SIZE_OF = ENCODE : aucune taille ne depend d'une adresse.
  --|  LI/LIF : QC_LEN(val) ; LCA/LSPA : QUAD_ADDR taille fixe (lui + addiw) ;
  --|  BRA : auipc + jalr (8) ; CALL : 24 fixes (adresse de retour auipc-relative,
  --|  micro-pile sp de 16 octets par appel, deroulee par EXC_RAISE).

  --|  Registres (codi tete) : s0 = x8 sommet de pile de travail, sp = x2 micro-pile
  --|  des retours (16 par appel), s1 = x9 display, s2 = x18 haut de co-pile,
  --|  s3 = x19 frame de co-pile, s4 = x20 haut du tas ; a0..a5 = x10..x15 ;
  --|  t0..t2 = x5..x7 scratch. Une instruction = un mot de 32 bits (DD).
  --|  Ada 83 n'a pas d'operateurs de bits : les champs d'un mot sont COMPOSES
  --|  PAR ADDITION (rd bits 7-11 = * 128, rs1 bits 15-19 = * 32768, rs2 et imm12
  --|  bits 20-31 = * 1048576, hi20 bits 12-31 = * 4096, imm[11:5] de S-type
  --|  bits 25-31 = * 33554432). 'and 0xFFF' = mod 4096 (complement a deux) ;
  --|  'shr 12' de fasmg est ARITHMETIQUE (plancher) : ( V - V mod 4096 ) / 4096.

			--
  procedure		DD		( V :SYMBOLS.VALUE_TYPE )					--| un mot d'instruction
  is			--
    U			: SYMBOLS.VALUE_TYPE	:= V;
  begin
    for  I in 1 .. 4  loop
      B( INTEGER( U mod 256 ) );
      U := ( U - ( U mod 256 ) ) / 256;
    end loop;

  end	DD;
	--

  --|  Immediat 12 bits signe (addi/ld) : lo12s dans -2048..2047 et hi20 compense
  --|  (lui/auipc) tels que hi20 * 4096 + lo12s = V (codi : lo12 = V and 0xFFF,
  --|  bit 11 -> lo12 - 0x1000 et hi20 = (V - lo12s) shr 12).

			-----
  function		LO12S		( V :SYMBOLS.VALUE_TYPE )		return SYMBOLS.VALUE_TYPE
  is			-----
    L			: constant SYMBOLS.VALUE_TYPE	:= V mod 4096;
  begin
    if  L >= 2048  then
      return  L - 4096;
    end if;
    return  L;
  end	LO12S;
	-----

			----
  function		HI20		( V :SYMBOLS.VALUE_TYPE )		return SYMBOLS.VALUE_TYPE
  is			----
  begin
    return  ( ( V - LO12S( V ) ) / 4096 ) mod 1048576;						--| and 0xFFFFF
  end	HI20;
	----

			-----
  function		IMM12		( V :SYMBOLS.VALUE_TYPE )		return SYMBOLS.VALUE_TYPE
  is			-----										--| (V and 0xFFF) shl 20 : I-type
  begin
    return  ( V mod 4096 ) * 1048576;
  end	IMM12;
	-----

			----
  function		SIMM		( V :SYMBOLS.VALUE_TYPE )		return SYMBOLS.VALUE_TYPE
  is			----										--| S-type : imm[11:5] bits 25-31, imm[4:0] bits 7-11
    I			: constant SYMBOLS.VALUE_TYPE	:= V mod 4096;
  begin
    return  ( I / 32 ) * 33554432 + ( I mod 32 ) * 128;
  end	SIMM;
	----

  --|  QUAD_CONST rd, val (codi) : addi (12 bits) ; lui + addiw si lo12 /= 0
  --|  (32 bits) ; sinon 32 bits hauts (addi, ou lui + addi si lo12 /= 0) puis
  --|  trois slli 11/11/10 chacun suivi d'un addi si le chunk est non nul.
  --|  Taille = fonction de val SEULE (contrat SIZE_OF : val est un immediat ou
  --|  un offset de P2, jamais une adresse - les adresses passent par QUAD_ADDR).

			------
  function		QC_LEN		( V :SYMBOLS.VALUE_TYPE )		return SYMBOLS.VALUE_TYPE
  is			------
    N			: SYMBOLS.VALUE_TYPE;
    LO32		: SYMBOLS.VALUE_TYPE;
    HI32		: SYMBOLS.VALUE_TYPE;
  begin
    if  V >= -2048  and then  V <= 2047  then
      return  4;
    elsif  V >= -2147483648  and then  V <= 2147483647  then
      if  LO12S( V ) = 0  then
        return  4;
      end if;
      return  8;
    end if;
    LO32 := V mod 4294967296;
    HI32 := ( V - LO32 ) / 4294967296;								--| hi32s (signe, plancher)
    if  HI32 >= -2048  and then  HI32 <= 2047  then
      N := 4;
    elsif  LO12S( HI32 ) = 0  then
      N := 4;
    else
      N := 8;
    end if;
    N := N + 12;											--| trois slli
    if  LO32 / 2097152 /= 0  then									--| ca = lo32 shr 21 (11 bits)
      N := N + 4;
    end if;
    if  ( LO32 / 1024 ) mod 2048 /= 0  then								--| cb
      N := N + 4;
    end if;
    if  LO32 mod 1024 /= 0  then									--| cc
      N := N + 4;
    end if;
    return  N;

  end	QC_LEN;
	------

			------------
  procedure		E_QUAD_CONST	( RD, V :SYMBOLS.VALUE_TYPE )
  is			------------
    RDF			: constant SYMBOLS.VALUE_TYPE	:= RD * 128 + RD * 32768;				--| rd et rs1 = rd
    LO32		: SYMBOLS.VALUE_TYPE;
    HI32		: SYMBOLS.VALUE_TYPE;
    C			: SYMBOLS.VALUE_TYPE;
  begin
    if  V >= -2048  and then  V <= 2047  then
      DD( 16#13# + IMM12( V ) + RD * 128 );								--| addi rd, x0, val
      return;
    elsif  V >= -2147483648  and then  V <= 2147483647  then
      DD( 16#37# + HI20( V ) * 4096 + RD * 128 );							--| lui rd, hi20
      if  LO12S( V ) /= 0  then
        DD( 16#1B# + IMM12( LO12S( V ) ) + RDF );							--| addiw rd, rd, lo12s
      end if;
      return;
    end if;
    LO32 := V mod 4294967296;
    HI32 := ( V - LO32 ) / 4294967296;
    if  HI32 >= -2048  and then  HI32 <= 2047  then
      DD( 16#13# + IMM12( HI32 ) + RD * 128 );								--| addi rd, x0, hi32s
    else
      DD( 16#37# + HI20( HI32 ) * 4096 + RD * 128 );							--| lui rd, hi20h
      if  LO12S( HI32 ) /= 0  then
        DD( 16#13# + IMM12( LO12S( HI32 ) ) + RDF );							--| addi rd, rd, lo12hs
      end if;
    end if;
    DD( 16#1013# + 11 * 1048576 + RDF );								--| slli rd, rd, 11
    C := LO32 / 2097152;
    if  C /= 0  then
      DD( 16#13# + C * 1048576 + RDF );									--| addi rd, rd, ca
    end if;
    DD( 16#1013# + 11 * 1048576 + RDF );								--| slli rd, rd, 11
    C := ( LO32 / 1024 ) mod 2048;
    if  C /= 0  then
      DD( 16#13# + C * 1048576 + RDF );									--| addi rd, rd, cb
    end if;
    DD( 16#1013# + 10 * 1048576 + RDF );								--| slli rd, rd, 10
    C := LO32 mod 1024;
    if  C /= 0  then
      DD( 16#13# + C * 1048576 + RDF );									--| addi rd, rd, cc
    end if;

  end	E_QUAD_CONST;
	------------

			-----------
  procedure		E_QUAD_ADDR	( RD, V :SYMBOLS.VALUE_TYPE )					--| adresse : TAILLE FIXE 8 (lui + addiw)
  is			-----------
  begin
    if  V < 0  or else  V > 16#7FFFFFFF#  then
      FAULT( "QUAD_ADDR : adresse hors 31 bits" );							--| assert du codi
    end if;
    DD( 16#37# + HI20( V ) * 4096 + RD * 128 );								--| lui rd, hi20
    DD( 16#1B# + IMM12( LO12S( V ) ) + RD * 128 + RD * 32768 );						--| addiw rd, rd, lo12
  end	E_QUAD_ADDR;
	-----------

			---
  procedure		E_B		( TARGET :SYMBOLS.VALUE_TYPE )					--| BRA : auipc t0, hi20 ; jalr x0, lo12(t0)
  is			---
    REL			: constant SYMBOLS.VALUE_TYPE	:= TARGET - ( ORG + SYMBOLS.VALUE_TYPE( TOP ) );	--| codi : lbl - $ + 8 = cible - adresse de l'auipc
  begin
    if  REL mod 4 /= 0  or else  REL < -2147483648  or else  REL > 2147483647  then
      FAULT( "BRA : deplacement hors plage" );								--| assert du codi
    end if;
    DD( 16#297# + HI20( REL ) * 4096 );									--| auipc t0, hi20
    DD( 16#28067# + IMM12( LO12S( REL ) ) );								--| jalr x0, lo12(t0)
  end	E_B;
	---

  --|  LOAD_QUAD ofs, Ra, Rd / STORE_QUAD ofs, Ra, Rs du codi : ld/sd imm12
  --|  (-2048..2047) ; sinon QUAD_CONST 5, ofs ; add t0, Ra, t0 ; ld/sd 0(t0).

			-----
  function		M_LEN		( D :SYMBOLS.VALUE_TYPE )		return SYMBOLS.VALUE_TYPE
  is			-----
  begin
    if  D >= -2048  and then  D <= 2047  then
      return  4;
    end if;
    return  QC_LEN( D ) + 8;
  end	M_LEN;
	-----

			----
  procedure		E_LQ		( OFS, RA, RD :SYMBOLS.VALUE_TYPE )				--| ld Rd, ofs(Ra)
  is			----
  begin
    if  OFS >= -2048  and then  OFS <= 2047  then
      DD( 16#3003# + IMM12( OFS ) + RA * 32768 + RD * 128 );
    else
      E_QUAD_CONST( 5, OFS );										--| t0 = ofs
      DD( 16#005002B3# + RA * 32768 );									--| add t0, Ra, t0
      DD( 16#0002B003# + RD * 128 );									--| ld Rd, 0(t0)
    end if;
  end	E_LQ;
	----

			----
  procedure		E_SQ		( OFS, RA, RS :SYMBOLS.VALUE_TYPE )				--| sd Rs, ofs(Ra)
  is			----
  begin
    if  OFS >= -2048  and then  OFS <= 2047  then
      DD( 16#3023# + SIMM( OFS ) + RS * 1048576 + RA * 32768 );
    else
      E_QUAD_CONST( 5, OFS );
      DD( 16#005002B3# + RA * 32768 );									--| add t0, Ra, t0
      DD( 16#0002B023# + RS * 1048576 );								--| sd Rs, 0(t0)
    end if;
  end	E_SQ;
	----

  --|  Display (codi FP_IN_RAX / RAX_IN_FP / FP_IN_RBP) : ld/sd 8*lvl(s1), lvl 0..31.

  procedure		E_FPA		( LVL :SYMBOLS.VALUE_TYPE )					--| ld a0, 8*lvl(s1)
  is			-----
  begin
    if  LVL < 0  or else  LVL > 31  then
      FAULT( "niveau de frame hors 0..31" );
    end if;
    E_LQ( 8 * LVL, 9, 10 );
  end	E_FPA;
	-----

			--------
  procedure		E_RAX_FP	( LVL :SYMBOLS.VALUE_TYPE )						--| sd a0, 8*lvl(s1)
  is			--------
  begin
    E_SQ( 8 * LVL, 9, 10 );
  end	E_RAX_FP;
	--------

			--------
  procedure		E_FP_RBP	( LVL :SYMBOLS.VALUE_TYPE )						--| ld s0, 8*lvl(s1)
  is			--------
  begin
    E_LQ( 8 * LVL, 9, 8 );
  end	E_FP_RBP;
	--------

			------
  function		S_BASE		( LVL :SYMBOLS.VALUE_TYPE )		return SYMBOLS.VALUE_TYPE
  is			------

    if  LVL = -1  then
      return  8;											--| POP_RAX : adresse empilee

  end	S_BASE;

  is			------

      E_POP_RAX;
    else
      E_FPA( LVL );
    end if;
  end	E_BASE;

  --|  Acces [a0 + disp] (FETCH_* / STORE_* du codi) : imm12 (-2048..2047) ;
  --|  sinon QUAD_CONST 5, disp ; add a0, a0, t0 ; acces a 0(a0). OP = le mot
  --|  de l'acces avec imm = 0 (rs1 = a0, rd = a0 ou rs2 = a1).
  --|  FETCH_QWORD / STORE_QWORD passent par LOAD_QUAD / STORE_QUAD (t0 base).

			------
  procedure		E_LOAD		( D, OP :SYMBOLS.VALUE_TYPE )
  is			------
  begin
    if  D >= -2048  and then  D <= 2047  then
      DD( OP + IMM12( D ) );
    else
      E_QUAD_CONST( 5, D );
      DD( 16#00550533# );										--| add a0, a0, t0
      DD( OP );
    end if;
  end	E_LOAD;
	------

			-------
  procedure		E_STORE		( D, OP :SYMBOLS.VALUE_TYPE )
  is			-------
  begin
    if  D >= -2048  and then  D <= 2047  then
      DD( OP + SIMM( D ) );
    else
      E_QUAD_CONST( 5, D );
      DD( 16#00550533# );										--| add a0, a0, t0
      DD( OP );
    end if;
  end	E_STORE;

  --|  Ajout d'un deplacement a a0 (LVA / LIVA) : rien si 0 ; addi imm12 ;
  --|  sinon QUAD_CONST 5 puis add a0, a0, t0.

			-----
  function		A_LEN		( D :SYMBOLS.VALUE_TYPE )		return SYMBOLS.VALUE_TYPE
  is			-----
  begin
    if  D = 0  then
      return  0;
    elsif  D >= -2048  and then  D <= 2047  then
      return  4;
    end if;
    return  QC_LEN( D ) + 4;
  end	A_LEN;
	-----

			-----
  procedure		E_ADD		( D :SYMBOLS.VALUE_TYPE )
  is			-----
  begin
    if  D = 0  then
      null;
    elsif  D >= -2048  and then  D <= 2047  then
      DD( 16#00050513# + IMM12( D ) );									--| addi a0, a0, disp
    else
      E_QUAD_CONST( 5, D );
      DD( 16#00550533# );										--| add a0, a0, t0
    end if;
  end	E_ADD;
	-----

  --|  LINK lvl, alloc (codi) : partage par LINK et ELB (alloc = loc_siz).
  --|  12 (co-pile) + 16 si lvl > 0 + selon alloc8 = 8*((alloc+7)/8) :
  --|  0 / 4 (addi) / QC_LEN + 4 (t0).

			--------
  function		LINK_LEN	( LVL, ALLOC :SYMBOLS.VALUE_TYPE )	return SYMBOLS.VALUE_TYPE
  is			--------

    A8			: constant SYMBOLS.VALUE_TYPE	:= 8 * ( ( ALLOC + 7 ) / 8 );
    N			: SYMBOLS.VALUE_TYPE		:= 12;

  begin
    if  LVL > 0  then
      N := N + 16;
    end if;
    if  A8 /= 0  then
      if  A8 <= 2047  then
        N := N + 4;
      else
        N := N + QC_LEN( A8 ) + 4;
      end if;
    end if;
    return  N;

  end	LINK_LEN;
	--------

			------
  procedure		E_LINK		( LVL, ALLOC :SYMBOLS.VALUE_TYPE )
  is			------
    A8			: constant SYMBOLS.VALUE_TYPE	:= 8 * ( ( ALLOC + 7 ) / 8 );
  begin
    if  ALLOC < 0  then
      FAULT( "LINK : alloc negatif" );									--| assert du codi
    end if;
    if  LVL > 0  then
      E_FPA( LVL );											--| a0 = ancien FP du niveau
      E_PUSH_RAX;
      E_SQ( 8 * LVL, 9, 8 );										--| sd s0, 8*lvl(s1)
    end if;
    if  A8 /= 0  then
      if  A8 <= 2047  then
        DD( 16#00040413# + IMM12( A8 ) );								--| addi s0, s0, alloc8
      else
        E_QUAD_CONST( 5, A8 );
        DD( 16#00540433# );										--| add s0, s0, t0
      end if;
    end if;
    DD( 16#01393023# );											--| sd s3, 0(s2)
    DD( 16#00090993# );											--| mv s3, s2
    DD( 16#00890913# );											--| addi s2, s2, 8

  end	E_LINK;
	------

			-------
  procedure		E_RTD		( P :SYMBOLS.VALUE_TYPE )
  is			-------
  begin
    if  P < 0  then
      FAULT( "RTD : prm_size negatif" );								--| assert du codi
    end if;
    if  P /= 0  then
      if  P <= 2047  then
        DD( 16#00040413# + IMM12( -P ) );								--| addi s0, s0, -prm_size
      else
        E_QUAD_CONST( 5, P );
        DD( 16#40540433# );										--| sub s0, s0, t0
      end if;
    end if;
    DD( 16#00013283# );											--| ld t0, 0(sp)
    DD( 16#01010113# );											--| addi sp, sp, 16
    DD( 16#00028067# );											--| jalr x0, 0(t0)
  end	E_RTD;
	-------

			-------
  function		RTD_LEN		( P :SYMBOLS.VALUE_TYPE )		return SYMBOLS.VALUE_TYPE
  is			-------
  begin
    if  P = 0  then
      return  12;
    elsif  P <= 2047  then
      return  16;
    end if;
    return  QC_LEN( P ) + 16;
  end	RTD_LEN;
	-------

			----------
  procedure		E_EXC_MACH	( LVL, CTX :SYMBOLS.VALUE_TYPE )				--| photographie de l'etat machine a [FP(lvl) + ctx]
  is			----------
  begin
    if  LVL < 0  or else  LVL > 31  then
      FAULT( "EXC_MACH : lvl hors 0..31 (assert du codi)" );
    end if;
    E_FPA( LVL );											--| a0 := FP(lvl)
    E_SQ( CTX + 16, 10, 8 );										--| pile de travail s0
    E_SQ( CTX + 24, 10, 2 );										--| micro-pile sp
    E_SQ( CTX + 32, 10, 19 );										--| frame co-pile s3
    E_SQ( CTX + 40, 10, 18 );										--| sommet co-pile s2
    E_QUAD_CONST( 6, LVL + 1 );										--| t1 = nlvl
    E_SQ( CTX + 48, 10, 6 );
    for  I in 0 .. INTEGER( LVL )  loop									--| prefixe du display FP(0..lvl)
      E_LQ( 8 * SYMBOLS.VALUE_TYPE( I ), 9, 6 );
      E_SQ( CTX + 56 + 8 * SYMBOLS.VALUE_TYPE( I ), 10, 6 );
    end loop;
  end	E_EXC_MACH;
	----------

			------------
  function		EXC_MACH_LEN	( LVL, CTX :SYMBOLS.VALUE_TYPE )	return SYMBOLS.VALUE_TYPE
  is			------------
    N			: SYMBOLS.VALUE_TYPE	:= 4 + QC_LEN( LVL + 1 );					--| FP_IN_RAX + QUAD_CONST 6, lvl+1
  begin
    N := N + M_LEN( CTX + 16 ) + M_LEN( CTX + 24 ) + M_LEN( CTX + 32 ) + M_LEN( CTX + 40 ) + M_LEN( CTX + 48 );
    for  I in 0 .. INTEGER( LVL )  loop
      N := N + M_LEN( 8 * SYMBOLS.VALUE_TYPE( I ) ) + M_LEN( CTX + 56 + 8 * SYMBOLS.VALUE_TYPE( I ) );
    end loop;
    return  N;
  end	EXC_MACH_LEN;
	------------

			-----------
  procedure		E_EXC_RAISE	( TOPD :SYMBOLS.VALUE_TYPE )					--| deroulage : POP avant dispatch
  is			-----------
  begin
    E_LQ( 0, 9, 11 );											--| a1 := FP(0)
    E_LQ( TOPD, 11, 10 );										--| a0 := contexte sommet
    DD( 16#00050693# );											--| mv a3, a0
    E_LQ( 0, 13, 12 );											--| a2 := PREV_CTX
    E_SQ( TOPD, 11, 12 );										--| POP avant dispatch
    E_LQ( 32, 13, 19 );											--| frame co-pile s3
    E_LQ( 40, 13, 18 );											--| sommet co-pile s2
    E_LQ( 24, 13, 6 );											--| t1 := sp sauvegarde
    DD( 16#00030113# );											--| mv sp, t1
    E_LQ( 48, 13, 12 );											--| a2 := NXT_LVL
    DD( 16#03868713# );											--| addi a4, a3, 56
    DD( 16#00048793# );											--| mv a5, s1
    DD( 16#00073283# );											--| boucle: ld t0, 0(a4)
    DD( 16#0057B023# );											--| sd t0, 0(a5)
    DD( 16#00870713# );											--| addi a4, a4, 8
    DD( 16#00878793# );											--| addi a5, a5, 8
    DD( 16#FFF60613# );											--| addi a2, a2, -1
    DD( 16#FE0616E3# );											--| bne a2, x0, boucle (-20)
    E_LQ( 16, 13, 8 );											--| pile de travail s0
    E_LQ( 8, 13, 5 );											--| t0 := DISPATCH
    DD( 16#00028067# );											--| jalr x0, 0(t0)
  end	E_EXC_RAISE;

----------------------------------------------------------------------------------------------------------------------
MODIFICATION 2 : SIZE_OF : table des tailles

ANCRE :
  function		SIZE_OF		( E :IR.ELT_ID )		return SYMBOLS.VALUE_TYPE
SUPPRIMER :
  is			-------
  begin
    FAULT( "hors tranche riscv64 : " & LEX.IMAGE( IR.MNEMO_OF( E ) ) );
    return 0;

REMPLACER PAR :
  is			-------

    M	:constant STRING	:= LEX.IMAGE( IR.MNEMO_OF( E ) );

  begin
    if  IR.KIND_OF( E ) /= IR.MACRO_CALL  then  return 0;							--| labels, virtual, map : zero octet
    end if;

    if  M = "DROP"  then return  4;
    elsif  M = "DUP"  then return  12;

--			L O A D	C O N S T A N T E S	  I M M E D I A T E S
    elsif  M = "LI"  then return  QC_LEN( OPV( E, 1, 0 ) ) + 8;						--| QUAD_CONST (1..8 mots) + PUSH_RAX
    elsif  M = "LIF"  then
      if  IR.N_OPS( E ) < 1  or else  IR.OP_TAG( E, 1 ) /= IR.FLT_OP  then
        FAULT( "LIF : litteral flottant attendu" );
      end if;
      return  QC_LEN( DOUBLE_BITS( IR.OP_FLT( E, 1 ) ) ) + 8;

--			L O A D	C O N S T A N T  /  V A R I A B L E   A D R E S S E
    elsif  M = "LCA"  then return  16;									--| QUAD_ADDR (8) + PUSH_RAX (8)
    elsif  M = "LSPA"  then return  16;
    elsif  M = "LVA"  then return  S_BASE( OPV( E, 1, -1 ) ) + A_LEN( OPV( E, 2, 0 ) ) + 8;
    elsif  M = "LIVA"  then return  S_BASE( OPV( E, 1, -1 ) ) + M_LEN( OPV( E, 2, 0 ) )
					+ A_LEN( OPV( E, 3, 0 ) ) + 8;

--				L O A D	D A T A		(unsigned/signed) : memes plages pour toutes les largeurs
    elsif  M = "ULB"  or else  M = "LB"  or else  M = "ULW"  or else  M = "LW"  or else  M = "ULD"  or else  M = "LD"
       or else  M = "LQ"  or else  M = "LA"  then
      return  S_BASE( OPV( E, 1, -1 ) ) + M_LEN( OPV( E, 2, 0 ) ) + 8;
    elsif  M = "ULIB"  or else  M = "LIB"  or else  M = "ULIW"  or else  M = "LIW"  or else  M = "ULID"  or else  M = "LID"
       or else  M = "LIQ"  or else  M = "LIA"  then
      return  S_BASE( OPV( E, 1, -1 ) ) + M_LEN( OPV( E, 2, 0 ) ) + M_LEN( OPV( E, 3, 0 ) ) + 8;

--				S T O R E	  D A T A
    elsif  M = "SB"  or else  M = "SW"  or else  M = "SD"  or else  M = "SQ"  or else  M = "SA"  then
      return  8 + S_BASE( OPV( E, 1, -1 ) ) + M_LEN( OPV( E, 2, 0 ) );
    elsif  M = "SIB"  or else  M = "SIW"  or else  M = "SID"  or else  M = "SIQ"  or else  M = "SIA"  then
      return  8 + S_BASE( OPV( E, 1, -1 ) ) + M_LEN( OPV( E, 2, 0 ) ) + M_LEN( OPV( E, 3, 0 ) );

--		O P E R A T I O N S	  L O G I Q U E S ,  B I T   F I E L D S ,  A R I T H M E T I Q U E
    elsif  M = "ET"  then return  20;
    elsif  M = "OU"  then return  20;
    elsif  M = "OUX"  then return  20;
    elsif  M = "SHL"  then return  20;
    elsif  M = "SHR"  then return  20;
    elsif  M = "SAR"  then return  20;
    elsif  M = "NON"  then return  12;
    elsif  M = "UBFX"  then return  48;
    elsif  M = "SBFX"  then return  48;
    elsif  M = "BFI"  then return  72;
    elsif  M = "DEC"  then return  12;
    elsif  M = "INC"  then return  12;
    elsif  M = "NEG"  then return  12;
    elsif  M = "ABS"  then return  20;
    elsif  M = "CLAMP0"  then return  20;
    elsif  M = "ADD"  then return  20;
    elsif  M = "SUB"  then return  20;
    elsif  M = "MUL"  then return  20;
    elsif  M = "DIV"  then return  28;
    elsif  M = "REMI"  then return  28;
    elsif  M = "MODI"  then return  44;

--	F L O T T A N T ,  C O N V E R S I O N S
    elsif  M = "FADD"  then return  20;
    elsif  M = "FSUB"  then return  20;
    elsif  M = "FMUL"  then return  20;
    elsif  M = "FDIV"  then return  20;
    elsif  M = "FNEG"  then return  12;
    elsif  M = "FABS"  then return  12;
    elsif  M = "FEXP"  then return  52;
    elsif  M = "CVTIF"  then return  12;
    elsif  M = "CVTFI"  then return  12;
    elsif  M = "CVTFIR"  then return  12;
    elsif  M = "CVTIX"  then return  168;
    elsif  M = "CVTXI"  then return  204;

--				C O M P A R A I S O N S
    elsif  M = "CEQ"  then return  24;
    elsif  M = "CNE"  then return  24;
    elsif  M = "CGT"  then return  20;
    elsif  M = "CGE"  then return  24;
    elsif  M = "CLT"  then return  20;
    elsif  M = "CLE"  then return  24;
    elsif  M = "FCEQ"  then return  20;
    elsif  M = "FCNE"  then return  24;
    elsif  M = "FCGT"  then return  20;
    elsif  M = "FCGE"  then return  20;
    elsif  M = "FCLT"  then return  20;
    elsif  M = "FCLE"  then return  20;

--			O P E R A T I O N S	  C O N T R O L E	D E   F L O T
    elsif  M = "BRA"  then return  8;									--| auipc + jalr, taille FIXE (TRAITS.BRA_SIZE)
    elsif  M = "BT"  or else  M = "BF"  then return  20;							--| POP_RAX + beq/bne +12 + BRA (forme longue)
    elsif  M = "CALL"  then return  24;									--| auipc, addi, addi sp, sd, BRA .elab (taille FIXE)
    elsif  M = "CALLI"  then return  28;								--| POP_RAX + auipc, addi, addi sp, sd, jalr a0
    elsif  M = "RTD"  then return  RTD_LEN( OPV( E, 1, 0 ) );

--			O P E R A T I O N S	  G E S T I O N   D E   P I L E
    elsif  M = "LINK"  then return  LINK_LEN( OPV( E, 1, 0 ), OPV( E, 2, 0 ) );
    elsif  M = "UNLINK"  then return  20;								--| FP_IN_RBP + POP_RAX + RAX_IN_FP + ld s3
    elsif  M = "UNLINKR"  then return  24;								--| ... + mv s2, s3
    elsif  M = "PRO"  then return  8;									--| BRA post
    elsif  M = "ELB"  then										--| LINK lvl, loc_siz (elab deja adresse en P2B)
      return  LINK_LEN( OPV( E, 1, 0 ), SYMBOLS.VALUE_OF( SYMBOLS.RESOLVE( "loc_siz" ) ) );
    elsif  M = "BEGIN_BLOC_DEF"  then return  8;								--| BRA IMAGES.skip (TRAITS.BRA_SIZE)
    elsif  M = "CO_VAR"  then return  32;
    elsif  M = "HEAP_ALLOC"  then return  36;
    elsif  M = "EXC_MACH"  then return  EXC_MACH_LEN( OPV( E, 1, 0 ), OPV( E, 2, 0 ) );
    elsif  M = "EXC_RAISE"  then return  76 + 2 * M_LEN( OPV( E, 1, 0 ) );					--| 19 mots fixes (boucle comprise) ; top lu et ecrit

--		O P E R A T I O N S	  L O G I Q U E S	D E   B L O C   (LRM 4.5.1)
    elsif  M = "BLKAND"  then return  60;
    elsif  M = "BLKOU"  then return  60;
    elsif  M = "BLKOUX"  then return  60;
    elsif  M = "BLKNOT"  then return  44;
    elsif  M = "BLKMOV"  then return  52;
    elsif  M = "BLKCMP"  then return  80;
    elsif  M = "LEXCMP"  then
      declare
	SIZC	:constant SYMBOLS.VALUE_TYPE	:= OPV( E, 1, 8 );
      begin
	if  SIZC /= 1  and then  SIZC /= 2  and then  SIZC /= 4  and then  SIZC /= 8  then
	  FAULT( "LEXCMP : taille de composant non supportee" );						--| err du codi
	end if;
	return  136;										--| 4 POP + 24 mots + PUSH, quelle que soit la paire de charges
      end;

--				SPECIFIQUE  L I N U X   RISC-V64
    elsif  M = "SYS_CLOCK_GETTIME"  then return  24;
    elsif  M = "SYS_PUT_CHAR"  then return  24;
    elsif  M = "SYS_PUT_STR"  then return  44;
    elsif  M = "SYS_GET_CHAR"  then return  132;
    elsif  M = "SYS_GET_STR"  then return  60;
    elsif  M = "SYS_FILE_CREATE"  then return  100;
    elsif  M = "SYS_FILE_OPEN"  then return  100;
    elsif  M = "SYS_FILE_SET_POS"  then return  40;
    elsif  M = "SYS_FILE_GET_POS"  then return  32;
    elsif  M = "SYS_FILE_GET_SIZE"  then return  80;
    elsif  M = "SYS_FILE_WRITE"  then return  48;
    elsif  M = "SYS_FILE_READ"  then return  48;
    elsif  M = "SYS_FILE_CLOSE"  then return  24;
    elsif  M = "SYS_FILE_DELETE"  then return  96;
    elsif  M = "SYS_EXIT"  then
      if  IR.N_OPS( E ) >= 1  and then  IR.OP_TAG( E, 1 ) = IR.INT_OP  and then  IR.OP_INT( E, 1 ) /= 0  then
        return  QC_LEN( IR.OP_INT( E, 1 ) ) + 8;
      end if;
      return  12;											--| addi a0, x0, 0 ; addi a7, x0, 93 ; ecall

    else
      FAULT( "hors table riscv64 : " & M );
      return 0;
    end if;


----------------------------------------------------------------------------------------------------------------------
MODIFICATION 3 : ENCODE : table des encodages
----------------------------------------------------------------------------------------------------------------------
ANCRE :
  procedure		ENCODE		( E :IR.ELT_ID )
SUPPRIMER :
  is			------
  begin
    FAULT( "hors tranche riscv64 : " & LEX.IMAGE( IR.MNEMO_OF( E ) ) );

REMPLACER PAR :
  is			------

    M			: constant STRING		:= LEX.IMAGE( IR.MNEMO_OF( E ) );
    V			: SYMBOLS.VALUE_TYPE;

  begin
    if  M = "DROP"  then
      DD( 16#FF840413# );								--| addi s0, s0, -8

    elsif  M = "DUP"  then
      DD( 16#00043503# );								--| ld  a0, 0(s0)  amener le TOS dans a0
      DD( 16#00A43423# );								--| sd  a0, 8(s0)  dupliquer
      DD( 16#00840413# );								--| addi s0, s0, 8

    elsif  M = "LI"  then
      if  IR.N_OPS( E ) < 1  then
        FAULT( "LI : operande attendu" );
      end if;
      E_QUAD_CONST( 10, OPV( E, 1, 0 ) );								--| litteral, nom ou EXPR (corpus : LI size*8)
      E_PUSH_RAX;

    elsif  M = "LIF"  then
      if  IR.N_OPS( E ) < 1  or else  IR.OP_TAG( E, 1 ) /= IR.FLT_OP  then
        FAULT( "LIF : litteral flottant attendu" );
      end if;
      E_QUAD_CONST( 10, DOUBLE_BITS( IR.OP_FLT( E, 1 ) ) );							--| motif IEEE 754 par QUAD_CONST
      E_PUSH_RAX;

    elsif  M = "LCA"  then
      if  IR.N_OPS( E ) < 1  or else  IR.OP_TAG( E, 1 ) /= IR.NAME_OP  then
        FAULT( "LCA : nom attendu" );
      end if;
      SYMBOLS.USE_SCOPE( IR.SCOPE_OF( E ) );
      V := SYMBOLS.VALUE_OF( SYMBOLS.RESOLVE( LEX.IMAGE( IR.OP_TXT( E, 1 ) ) ) );
      if  IR.N_OPS( E ) >= 2  and then  IR.OP_TAG( E, 2 ) = IR.INT_OP  then
	V := V + IR.OP_INT( E, 2 );
      elsif  IR.N_OPS( E ) >= 2  and then  IR.OP_TAG( E, 2 ) = IR.EXPR_OP  then
	V := V + LEX.EVAL( IR.OP_TXT( E, 2 ) );
      end if;
      E_QUAD_ADDR( 10, V );										--| ptr + disp
      E_PUSH_RAX;

    elsif  M = "LSPA"  then
      V := ELAB_TARGET( E );
      E_QUAD_ADDR( 10, V );										--| .elab (forme canonique)
      E_PUSH_RAX;

    elsif  M = "LVA"  then
      E_BASE( OPV( E, 1, -1 ) );
      E_ADD( OPV( E, 2, 0 ) );
      E_PUSH_RAX;

    elsif  M = "LIVA"  then
      E_IBASE( OPV( E, 1, -1 ), OPV( E, 2, 0 ) );
      E_ADD( OPV( E, 3, 0 ) );
      E_PUSH_RAX;

--			L O A D	D A T A
    elsif  M = "ULB"  then
      E_BASE( OPV( E, 1, -1 ) );  E_FETCH_UB( OPV( E, 2, 0 ) );  E_PUSH_RAX;
    elsif  M = "ULW"  then
      E_BASE( OPV( E, 1, -1 ) );  E_FETCH_UW( OPV( E, 2, 0 ) );  E_PUSH_RAX;
    elsif  M = "ULD"  then
      E_BASE( OPV( E, 1, -1 ) );  E_FETCH_UD( OPV( E, 2, 0 ) );  E_PUSH_RAX;
    elsif  M = "LB"  then
      E_BASE( OPV( E, 1, -1 ) );  E_FETCH_B( OPV( E, 2, 0 ) );  E_PUSH_RAX;
    elsif  M = "LW"  then
      E_BASE( OPV( E, 1, -1 ) );  E_FETCH_W( OPV( E, 2, 0 ) );  E_PUSH_RAX;
