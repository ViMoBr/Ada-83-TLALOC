-- expander-declarations-types_decls.adb
-- Le calcul Ada de la taille statique d'un record doit SIMULER l'alignement
-- que STATOFS applique reellement (align_w/align_d/align_q selon le 3e
-- argument, deja fourni par STATIC_TYPE_ALIGN_BYTES).  Sans cela,
-- CD_IMPL_SIZE < size reel des qu'un champ aligne cree du padding
-- (temoin : RPG_DATA, DATA access aligne 8 -> offset 16, size 24, mais
-- CD_IMPL_SIZE = 128 bits -> stride d'agregat 16 -> chaque element
-- ecrase le DATA du precedent).

-- 1) Helper commun, a placer pres de la spec forward de
--    STATIC_TYPE_ALIGN_BYTES (l. 960) :

			-----------------
    procedure		ALIGN_STATIC_BITS	( CUMUL : in out NATURAL;  COMP_TYPE : TREE )
    is			-----------------
    -- Arrondit le cumul de bits a l'alignement du champ qui va etre pose,
    -- MIROIR EXACT des align_* que STATOFS emet avec le meme
    -- STATIC_TYPE_ALIGN_BYTES.  Toute divergence entre ce calcul et le
    -- layout fasmg est un bug d'ecrasement memoire (cf. RPG_DATA).
      A_BITS	: NATURAL	:= STATIC_TYPE_ALIGN_BYTES( COMP_TYPE ) * CODI.STORAGE_UNIT;
    begin
      if  A_BITS > 0  then
        CUMUL := ( ( CUMUL + A_BITS - 1 ) / A_BITS ) * A_BITS;
      end if;
    end	ALIGN_STATIC_BITS;
	-----------------

-- 2) TRAITER_LES_DISCRIMINANTS, l. 1583 : inserer l'appel AVANT l'addition.

	    ALIGN_STATIC_BITS( STATIC_SIZE, DISCR_TYPE );
	    STATIC_SIZE := STATIC_SIZE
		+ CODI.STORAGE_UNIT * ( ( DISCR_SIZE + CODI.STORAGE_UNIT - 1 ) / CODI.STORAGE_UNIT );

-- 3) TRAITER_COMP_LIST / CHAMPS_REGULIERS, l. 1647 : idem.

	          ALIGN_STATIC_BITS( STATIC_SIZE, COMP_TYPE );
	          STATIC_SIZE := STATIC_SIZE
			   + CODI.STORAGE_UNIT * ( ( COMP_SIZE + CODI.STORAGE_UNIT - 1 ) / CODI.STORAGE_UNIT );

-- 4) STATIC_RECORD_SIZE_BITS : memes insertions dans ADD_DISCRIMINANTS
--    (avant SIZE := SIZE + ROUND_STORAGE_BITS( DISCR_SIZE )) et dans
--    ADD_COMP_LIST (avant SIZE := SIZE + ROUND_STORAGE_BITS( COMP_SIZE )) :

	      ALIGN_STATIC_BITS( SIZE, DISCR_TYPE );
	      SIZE := SIZE + ROUND_STORAGE_BITS( DISCR_SIZE );
	      ...
	        ALIGN_STATIC_BITS( SIZE, COMP_TYPE );
	        SIZE := SIZE + ROUND_STORAGE_BITS( COMP_SIZE );

-- NOTES
-- a) Pas d'arrondi FINAL : STATOFS n'emet pas d'align apres le dernier
--    champ, "size = $" tombe pile sur la fin du dernier rb.  Le miroir
