PATCH GOTO v1 — DN_GOTO / DN_LABELED dans l'expander
========================================================================
Perimetre v1 (consigne) :
  - goto ARRIERE : general (denivele UNLINK par niveau + EXC_POP des
    blocs proteges traverses, forme de CODE_EXIT, pieges n 69 et 34) ;
  - goto AVANT meme niveau : BRA nu (motif ADA_COMP, fasmg resout) ;
  - goto AVANT inter-niveaux : REFUS BRUYANT a l'expansion (piege n 53),
    verifie au moment ou l'etiquette est atteinte.
Formes confirmees au dump GOTO_DUMP (piege n 71) :
  - DN_GOTO.as_name = DN_USED_NAME_ID, SM_DEFN -> DN_LABEL_ID ;
  - DN_LABELED.as_source_name_s = liste de DN_LABEL_ID, as_stm = stm ;
  - CD_LABEL des label_id VIERGE apres compilation -> ON NE LE LIT NI
    NE L'ECRIT : table CODI indexee par le noeud TREE (egalite "=").
Cinq blocs, dans l'ordre. Relire chaque ancre avant collage (piege n 66).

			-- GOTO SERVICE (etiquettes <<L>> et instruction goto, LRM 5.9)
			--| Table par CORPS (les etiquettes ne franchissent pas les corps,
			--| LRM 5.9), discipline de pile pour l'imbrication des corps.
			--| Indexee par le NOEUD DN_LABEL_ID (le CD_LABEL des label_id est
			--| vierge -- constat au dump GOTO_DUMP -- on ne le touche pas).

    MAX_GOTO_LABELS		: constant		:= 64;					--| NB MAX D'ETIQUETTES GOTO PAR IMBRICATION DE CORPS
    subtype GOTO_LBL_IDX	is NATURAL		range 0 .. MAX_GOTO_LABELS;

    type GOTO_LBL_REC		is record
      ID		: TREE;								--| le DN_LABEL_ID (cle, egalite TREE)
      LBL		: LABEL_TYPE;							--| etiquette FASM partagee goto/labeled
      DEFINED	: BOOLEAN;							--| l'etiquette a ete EMISE
      LEVEL	: LEVEL_NUM;							--| niveau d'emission (valide si DEFINED)
      PEND_LVL	: INTEGER;							--| niveau des gotos en avant deja emis (-1 : aucun)
    end record;

    GOTO_LABELS		: array( 1 .. MAX_GOTO_LABELS ) of GOTO_LBL_REC;
    GOTO_LBL_TOP		: GOTO_LBL_IDX	:= 0;
    GOTO_BODY_BASE		: GOTO_LBL_IDX	:= 0;					--| base du corps courant dans la table

    function  GOTO_LABEL_ENTRY	( LABEL_ID :TREE )		return GOTO_LBL_IDX;	--| trouve ou cree l'entree du corps courant
    procedure GOTO_CHECK_BODY_END;							--| ceinture bruyante : goto en avant orphelin

			--^^^^^^^^^--
  function		  GOTO_LABEL_ENTRY	( LABEL_ID :TREE )	return GOTO_LBL_IDX
  is			-------------
  begin
    for  I in GOTO_BODY_BASE + 1 .. GOTO_LBL_TOP  loop
      if  GOTO_LABELS( I ).ID = LABEL_ID  then
        return I;
      end if;
    end loop;

    if  GOTO_LBL_TOP = MAX_GOTO_LABELS  then						-- refus bruyant (piege n 53)
      PUT_LINE( "; !!! GOTO_LABEL_ENTRY : table pleine (MAX_GOTO_LABELS)" );
      raise PROGRAM_ERROR;
    end if;

    GOTO_LBL_TOP := GOTO_LBL_TOP + 1;
    GOTO_LABELS( GOTO_LBL_TOP ) := ( ID	=> LABEL_ID,
			         LBL	=> NEW_LABEL,
			         DEFINED	=> FALSE,
			         LEVEL	=> 0,
			         PEND_LVL	=> -1 );
    return GOTO_LBL_TOP;

  end	GOTO_LABEL_ENTRY;
	----------------


			--^^^^^^^^^--
  procedure		  GOTO_CHECK_BODY_END
  is			-------------
  begin
    for  I in GOTO_BODY_BASE + 1 .. GOTO_LBL_TOP  loop
      if  not GOTO_LABELS( I ).DEFINED  then						-- un goto en avant a ete emis, l'etiquette
        PUT_LINE( "; !!! GOTO : etiquette jamais emise dans ce corps" );			-- jamais atteinte : sem le garantit,
        raise PROGRAM_ERROR;							-- ceinture bruyante (piege n 53)
      end if;
    end loop;

  end	GOTO_CHECK_BODY_END;

  procedure			CODE_LABELED		( LABELED :TREE )
  is
    LBL_SEQ	: SEQ_TYPE	:= LIST( D( AS_SOURCE_NAME_S, LABELED ) );
    LBL_ID	: TREE;
  begin
    while  not IS_EMPTY( LBL_SEQ )  loop
      POP( LBL_SEQ, LBL_ID );
      declare
        E	: CODI.GOTO_LBL_IDX	:= CODI.GOTO_LABEL_ENTRY( LBL_ID );
      begin
        if  CODI.GOTO_LABELS( E ).DEFINED  then						-- sem garantit l'unicite (LRM 5.1) ;
	PUT_LINE( "; !!! CODE_LABELED : etiquette emise deux fois" );			-- ceinture bruyante (piege n 53)
	raise PROGRAM_ERROR;
        end if;

        if  CODI.GOTO_LABELS( E ).PEND_LVL >= 0
	and then  CODI.GOTO_LABELS( E ).PEND_LVL /= CODI.CUR_LEVEL
        then										-- le BRA nu deja emis n'a fait ni ses
	PUT_LINE( "; !!! CODE_LABELED : goto en avant inter-niveaux non implante" );		-- UNLINK ni ses EXC_POP : refus bruyant
	raise PROGRAM_ERROR;							-- (perimetre v1 ; piege n 53)
        end if;

        CODI.GOTO_LABELS( E ).DEFINED := TRUE;
        CODI.GOTO_LABELS( E ).LEVEL   := CODI.CUR_LEVEL;
        PUT_LINE( LABEL_STR( CODI.GOTO_LABELS( E ).LBL ) & ':' );
      end;
    end loop;

    CODE_STM( D( AS_STM, LABELED ) );							-- l'instruction etiquetee elle-meme

  end	CODE_LABELED;

  procedure			CODE_GOTO			( ADA_GOTO :TREE )
  is
    TARGET	: TREE	:= D( AS_NAME, ADA_GOTO );
  begin
    if  TARGET.TY /= DN_LABEL_ID  then
      TARGET := D( SM_DEFN, TARGET );							-- forme du dump GOTO_DUMP :
    end if;										-- DN_USED_NAME_ID -> SM_DEFN -> DN_LABEL_ID
    if  TARGET.TY /= DN_LABEL_ID  then
      PUT_LINE( "; !!! CODE_GOTO : cible non resolue " & NODE_NAME'IMAGE( TARGET.TY ) );		-- refus bruyant (piege n 53)
      raise PROGRAM_ERROR;
    end if;

    declare
      E	: CODI.GOTO_LBL_IDX	:= CODI.GOTO_LABEL_ENTRY( TARGET );
    begin
      if  CODI.GOTO_LABELS( E ).DEFINED  then						-- GOTO ARRIERE : deniveler, forme de
        for  L in reverse CODI.GOTO_LABELS( E ).LEVEL + 1 .. CODI.CUR_LEVEL  loop			-- CODE_EXIT (pieges n 69 et 34)
	if  CODI.HANDLER_CTX_AT( L )  then  CODI.EXC_POP;  end if;				-- pop des blocs proteges traverses
	PUT_LINE( tab & "UNLINK" & LEVEL_NUM'IMAGE( L ) );
        end loop;

      else										-- GOTO AVANT : BRA nu, verification
        if  CODI.GOTO_LABELS( E ).PEND_LVL < 0  then					-- differee a CODE_LABELED (perimetre v1)
	CODI.GOTO_LABELS( E ).PEND_LVL := CODI.CUR_LEVEL;
        elsif  CODI.GOTO_LABELS( E ).PEND_LVL /= CODI.CUR_LEVEL  then
	PUT_LINE( "; !!! CODE_GOTO : gotos en avant de niveaux differents vers la meme etiquette" );
	raise PROGRAM_ERROR;							-- au plus un des deux serait juste
        end if;
      end if;

      PUT_LINE( tab & "BRA" & tab & LABEL_STR( CODI.GOTO_LABELS( E ).LBL ) );

    end;
  end	CODE_GOTO;

NOTES DE RELECTURE
------------------------------------------------------------------------
1. Surete du BRA intra-frame : rbp invariant aux frontieres
   d'instruction (VARzone allouee en bloc au LINK), pile d'operandes
   vide, R14 monotone (piege n 70 : un goto qui saute des CO_VAR fuit,
   il ne casse pas -- doctrine consignee).
2. Le goto ENTRANT dans une sequence protegee ou un handler est
   illegal (LRM 5.9) et refuse par le frontend ; rien a empiler,
   seulement a depiler -- la boucle ARRIERE s'en charge.
3. La verification de niveau des gotos en avant est une CEINTURE, pas
   une preuve : deux blocs freres de meme niveau seraient acceptes,
   mais sem refuse cette forme (LRM 5.9, cible englobante requise).
4. Temoin : GOTO_TEST.adb (joint), 4 sections auto-jugees ; le cas
   refuse (goto avant sortant d'un bloc) est en commentaire de queue,
   a compiler A PART pour verifier le refus bruyant.
5. Apres integration : rebuild hote (gnat), puis GOTO_TEST, puis
   recompilation d'ADA_COMP par TLALOC, puis dis_bonjour en option S.
