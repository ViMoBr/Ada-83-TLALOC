-- ============================================================================
-- CONVENTION (a inscrire dans CONVENTIONS_ARCHITECTURE.md) :
--   CODE_EXP d'une expression composite laisse
--     @DOUBLET pour un objet ENTIER (DN_USED_OBJECT_ID) ou un APPEL DE
--              FONCTION (DN_FUNCTION_CALL) ;
--     @DATA nue pour toute REFERENCE DE COMPOSANT (DN_INDEXED, DN_SELECTED,
--              DN_ALL, ...).
--   Tout consommateur qui veut l'@data DOIT passer par le helper ci-dessous.
--   Un "La ,0" inconditionnel apres CODE_EXP est un bug (5 occurrences
--   corrigees : egalite, actual indexe, return, agregat x3, init decl).
-- ============================================================================

-- 1) HELPER -- expander-expressions.adb, a declarer dans la spec du
--    paquetage EXPRESSIONS (visible d'instructions et declarations) :

			---------------------------
    procedure		CODE_COMPOSITE_DATA_ADDRESS	( EXP : TREE );
			---------------------------

--    corps, a placer pres d'OPERAND_DATA_ADDRESS dont il generalise la
--    discrimination :

			---------------------------
    procedure		CODE_COMPOSITE_DATA_ADDRESS	( EXP : TREE )
    is			---------------------------
    -- Compile EXP (composite) et laisse son @DATA sur la pile, quelle que
    -- soit la forme de EXP.  Discrimination d'OPERAND_DATA_ADDRESS
    -- (CODE_RECORD_EQUALITY) erigee en regle unique.
    begin
      CODE_EXP( EXP );
      if  EXP.TY = DN_USED_OBJECT_ID  or else  EXP.TY = DN_FUNCTION_CALL  then
        PUT_LINE( tab & "La  ,  0" );					-- @doublet -> data_ptr
      end if;								-- sinon : deja @data
    end	CODE_COMPOSITE_DATA_ADDRESS;
	---------------------------

-- 2) LES QUATRE SITES -- remplacer chaque paire
--        CODE_EXP( <exp> );          [ou EXPRESSIONS.CODE_EXP]
--        PUT_LINE( tab & "La  ,  0" );
--    par
--        CODE_COMPOSITE_DATA_ADDRESS( <exp> );

--   a) expander-expressions.adb ~5266-5267 (agregat record, COMP_EXP,
--      branche DN_RECORD) -- le segfault INSERT / NEXT => S.NEXT ;

--      porte le meme motif ;
--   b) expander-expressions.adb ~5488-5490 (agregat record, FIELD_EXP,
--      branche DN_RECORD) ;
--   c) expander-expressions.adb ~5607-5609 (composant record du parent) ;
--   d) expander-declarations.adb ~1120-1121 (init de declaration d'objet
--      record, INIT_EXP).

-- 3) REPRISE DES SITES DEJA CORRIGES A LA MAIN (facultatif, coherence) :
--    CODE_RETURN branche record (instructions ~1046) peut appeler le helper
--    au lieu de sa discrimination locale ; OPERAND_DATA_ADDRESS de
--    CODE_RECORD_EQUALITY peut lui deleguer.

-- NOTES
-- a) NE PAS l'appliquer aux "La ,0" de DESTINATION (result__ofs, doublets
--    anonymes SELARG/INDARG/ANON) : la leurs operandes sont des @doublets
--    par construction.
