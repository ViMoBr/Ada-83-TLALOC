LIVRAISON codi_riscv64.finc v1 -> v2 : mise au niveau de codi_x86_64.finc (reference) et codi_arm64.finc
======================================================================================================================
Format : une modification = ANCRE (ligne existante unique, inchangee) ; SUPPRIMER = bloc qui SUIT
immediatement l'ancre (vide = insertion) ; REMPLACER PAR = bloc pose a la place. Tabulations preservees.
Appliquer les commits dans l'ordre (les ancres de C-R4 sont posees par le texte d'origine, pas par C-R1).

######################################################################################################################
ORACLE C-R1 :
  1. fasmg sur TC_TEST07/09 riscv (jumeaux des temoins x86 : CALL/CALLI/RTD, EXC_MACH/EXC_RAISE) puis

  2. Temoin decisif : EXC_TEST1 compile pour riscv64 -- un raise dans G appele par F porteur du handler,

MODIFICATION 1 : tete : convention des registres (ra n'est plus utilise)
----------------------------------------------------------------------------------------------------------------------
ANCRE :
;	  RDI -> a5  (x15)  param syscall
SUPPRIMER :
;	  LR  -> ra  (x1)   link register, pousse manuellement sur sp pour CALL/RET
REMPLACER PAR :
;	  ra (x1) n'est PAS utilise par CALL/RTD : la LLIR maintient une micro-pile explicite
;	  d'adresses de retour sur sp (16 octets par appel), comme x30 sur arm64 -- c'est ce qui
;	  permet a EXC_RAISE de derouler les retours des frames abandonnes en restaurant sp.

SUPPRIMER :

REMPLACER PAR :
	; effectivement a stocker l'adresse de retour, l'autre est du padding d'alignement).

----------------------------------------------------------------------------------------------------------------------
MODIFICATION 3 : CALL
----------------------------------------------------------------------------------------------------------------------
ANCRE :
macro			CALL	prefix, subname			; CALL direct longue portee
SUPPRIMER :
;			-------------------------
; Sauvegarde ra sur la micro-pile 16-octets alignee, puis AUIPC+JALR.
  postpone
    if ~definite prefix#subname#_
      prefix#subname#_ = prefix#subname#.elab
    end if
  end postpone

  dd 0xFF010113							; addi sp, sp, -16
  dd 0x00113023							; sd ra, 0(sp)

  local rel, lo12, hi20
  lo12 = rel and 0xFFF
  if lo12 and 0x800
    hi20 = ((rel - (lo12 - 0x1000)) shr 12) and 0xFFFFF
    lo12 = lo12 - 0x1000
  else
    hi20 = (rel shr 12) and 0xFFFFF
  end if
  dd 0x00000297 or (hi20 shl 12)					; auipc t0, hi20
  dd 0x000280E7 or ((lo12 and 0xFFF) shl 20)			; jalr ra, lo12(t0)
  rel = prefix#subname#_ - $ + 8					; PC de l'AUIPC = $-8

  dd 0x00013083							; ld ra, 0(sp)
  dd 0x01010113							; addi sp, sp, 16
end macro
REMPLACER PAR :
;			-------------------------
; RISC-V n'empile pas l'adresse de retour. Pour conserver EXACTEMENT le modele
; LLIR/x86 (et arm64), on gere explicitement la micro-pile par sp : chaque appel
; reserve 16 octets (alignement), range l'adresse de retour a [sp], puis branche.
; RTD depile cette adresse. Ainsi une restauration de sp par EXC_RAISE deroule
; aussi les retours des frames abandonnes, sans avoir a sauver ra dans le contexte.
; (L'ancienne forme 'sd ra / jalr ra / ld ra' laissait ra dans un etat quelconque
; apres un deroulage : le RTD du frame porteur du handler retournait dans le vide.)
; L'adresse de retour est PC-relative (auipc t0, 0 ; addi t0, t0, 24 : la 6e
; instruction apres l'auipc, le BRA en occupant deux) : taille FIXE de 24 octets
; par CALL, aucun label,
; aucune dependance a une valeur de passe precedente (contrat SIZE_OF = ENCODE).
  postpone
    if ~definite prefix#subname#_
      prefix#subname#_ = prefix#subname#.elab
    end if
  end postpone

  dd 0x00000297							; auipc t0, 0		(t0 = adresse de cet auipc)
  dd 0x01828293							; addi t0, t0, 24	(adresse de retour = apres le jalr du BRA)
  dd 0xFF010113							; addi sp, sp, -16
  dd 0x00513023							; sd t0, 0(sp)
  BRA prefix#subname#_						; auipc t1 + jalr x0 (8 octets, +/-2 GiB)
end macro

MODIFICATION 4 : CALLI
----------------------------------------------------------------------------------------------------------------------
ANCRE :
macro			CALLI					; CALL indirect par valeur empilee
SUPPRIMER :
;			-----
;	Depile l'adresse cible dans a0, sauvegarde ra sur sp, jalr ra, a0, restaure ra.
  POP_RAX								; a0 = adresse cible
  dd 0xFF010113							; addi sp, sp, -16
  dd 0x00113023							; sd ra, 0(sp)
  dd 0x000500E7							; jalr ra, 0(a0)	(opcode 0x67, funct3=0, rs1=10, rd=1)
  dd 0x00013083							; ld ra, 0(sp)
  dd 0x01010113							; addi sp, sp, 16
end macro
REMPLACER PAR :
;			-----
;	Depile l'adresse cible dans a0, empile l'adresse de retour sur la micro-pile sp, saute.
  POP_RAX								; a0 = adresse cible
  dd 0x00000297							; auipc t0, 0		(t0 = adresse de cet auipc)
  dd 0x01428293							; addi t0, t0, 20	(adresse de retour = apres le jalr)
  dd 0xFF010113							; addi sp, sp, -16
  dd 0x00513023							; sd t0, 0(sp)
  dd 0x00050067							; jalr x0, 0(a0)
end macro

----------------------------------------------------------------------------------------------------------------------
MODIFICATION 5 : RTD
----------------------------------------------------------------------------------------------------------------------
ANCRE :
macro			RTD	param_size:0
SUPPRIMER :
;			--------------------
;	Desallouer param_size octets de la pile de travail puis retour (ret = jalr x0, 0(ra)).
;	L'adresse de retour est dans ra ; le 'CALL' du caller s'occupera de
;	restaurer le ra du niveau encore au-dessus.
  prm_size = param_size						; evaluer l'argument textuel (prm_siz-8) AVANT toute negation : and prime sur + - en fasmg
  assert prm_size >= 0
  if prm_size <> 0
    if prm_size <= 2047
;	addi s0, s0, -prm_size
      dd 0x00040413 or (((-prm_size) and 0xFFF) shl 20)
    else
      QUAD_CONST	5, prm_size					; t0 = prm_size
      dd 0x40540433							; sub s0, s0, t0  (funct7=0x20, rs1=8, rs2=5, rd=8)
    end if
  end if
  dd 0x00008067							; ret  (jalr x0, 0(ra))
end macro
REMPLACER PAR :
;			--------------------
;	Desallouer param_size octets de la pile de travail puis retour par la micro-pile sp
;	(adresse de retour rangee par CALL/CALLI ; 16 octets par appel).
  prm_size = param_size						; evaluer l'argument textuel (prm_siz-8) AVANT toute negation : and prime sur + - en fasmg
  assert prm_size >= 0
  if prm_size <> 0
    if prm_size <= 2047
;	addi s0, s0, -prm_size
      dd 0x00040413 or (((-prm_size) and 0xFFF) shl 20)
    else
      QUAD_CONST	5, prm_size					; t0 = prm_size
      dd 0x40540433							; sub s0, s0, t0  (funct7=0x20, rs1=8, rs2=5, rd=8)
    end if
  end if
  dd 0x00013283							; ld t0, 0(sp)		adresse de retour LLIR
  dd 0x01010113							; addi sp, sp, 16
  dd 0x00028067							; jalr x0, 0(t0)
end macro

----------------------------------------------------------------------------------------------------------------------
MODIFICATION 1 : QUAD_ADDR (nouvelle macro, apres QUAD_CONST)
----------------------------------------------------------------------------------------------------------------------
ANCRE :
      dd 0x00000013 or ((cc and 0xFFF) shl 20) or (rd shl 15) or (rd shl 7)	; addi rd, rd, cc
    end if
  end if
end macro
SUPPRIMER :
(rien : insertion apres l'ancre)
REMPLACER PAR :

;			..........................
macro			QUAD_ADDR	rd, val
;			..........................
;	Materialise une ADRESSE (symbole, reference avant possible) dans rd.
;	TAILLE FIXE : lui + addiw, TOUJOURS deux instructions (l'image ELF est chargee
;	sous 2 GiB : org 0x400000, code + constantes). L'addiw re-etend le signe du
;	resultat 32 bits, ce qui rend la compensation hi20 = 0x80000 inoffensive.
;	L'assert garde l'hypothese ; QUAD_CONST reste reservee aux IMMEDIATS
;	(taille f(val), pieges 88 et 157 : jamais de taille dependant d'une valeur
;	de la passe precedente).
  assert (val) >= 0 & (val) <= 0x7FFFFFFF
  local lo12a, hi20a
  lo12a = (val) and 0xFFF
  if lo12a and 0x800
    hi20a = (((val) - (lo12a - 0x1000)) shr 12) and 0xFFFFF
    lo12a = lo12a - 0x1000
  else
    hi20a = ((val) shr 12) and 0xFFFFF
  end if
  dd 0x00000037 or (hi20a shl 12) or (rd shl 7)				; lui rd, hi20
  dd 0x0000001B or ((lo12a and 0xFFF) shl 20) or (rd shl 15) or (rd shl 7)	; addiw rd, rd, lo12
end macro

MODIFICATION 1 : UNLINK : commentaire du contrat d'evasion, et UNLINKR
----------------------------------------------------------------------------------------------------------------------
ANCRE :
macro			UNLINK	lvl				; DISPOSE VARS LOCALES ET RESTORE FRAME PTR
SUPPRIMER :
;			-------------
  FP_IN_RBP	lvl						; throw locals : s0 = FP(lvl)
  POP_RAX								; depiler l'ancien FP sauvegarde
  RAX_IN_FP	lvl						; restaurer l'entree lvl du display
;	Liberer le frame de co-pile : ld s3, 0(s3)
  dd 0x0009B983							; ld s3, 0(s3)  (rs1=19, rd=19)
end macro
REMPLACER PAR :
;			-------------
  FP_IN_RBP	lvl						; throw locals : s0 = FP(lvl)
  POP_RAX								; depiler l'ancien FP sauvegarde
  RAX_IN_FP	lvl						; restaurer l'entree lvl du display
;	La co-pile N'EST PAS rendue ici (n 147, contrat d'evasion) : les donnees d'un
;	resultat de fonction TABLEAU vivent dans la region de ce frame et sont consommees
;	par l'appelant APRES le retour. La variante UNLINKR ci-dessous rend la co-pile ;
;	c'est l'EXPANDER qui choisit (EXIT_UNLINK_MNEMONIC) selon le type de resultat.
  dd 0x0009B983							; ld s3, 0(s3)  (rs1=19, rd=19)
end macro

;			--------------
macro			UNLINKR	lvl				; UNLINK + LIBERATION DE LA CO-PILE (s2 redescend a la cellule LINK de ce frame)
;			--------------
  FP_IN_RBP	lvl						; throw locals : s0 = FP(lvl)
  POP_RAX								; depiler l'ancien FP sauvegarde
  RAX_IN_FP	lvl						; restaurer l'entree lvl du display
;	Chantier co-pile (n 163) : rendre TOUT ce que ce frame a alloue, cellule LINK comprise.
;	INTERDIT pour un frame de fonction a resultat TABLEAU (contrat d'evasion n 147) :
;	l'expander n'emet UNLINKR que si le resultat est copie chez l'appelant avant l'epilogue.
  dd 0x00098913							; mv s2, s3	(sommet := cellule LINK de ce frame = s2 d'avant LINK)
  dd 0x0009B983							; ld s3, 0(s3)	(frame de co-pile anterieur)
end macro

# COMMIT C-R4  CVTIX / CVTXI sur produit 128 bits (SDIV128_64_POS, parite x86-64 imul/idiv et arm64 smulh)
######################################################################################################################
ORACLE C-R4 :
  1. FIX_TEST1 (15/15), FLOAT_FIXED_IO_TEST (11 sections), TEST_CALENDAR sous qemu-riscv64 : ce sont les temoins
     de la reference x86 pour CVTIX/CVTXI, y compris le gardien -2.5 en small 3/4 -> -2.25 (troncature vers zero).
  2. Temoin de debordement 64 bits (celui qui distingue l'ancienne forme mul/div de la forme 128 bits) :
     DURATION'LAST * 1 via CVTIX avec DENOM tel que I*DENOM > 2**63 -- valeur attendue = celle du binaire x86.
  3. Verification unitaire hors cible deja faite ici : SDIV128_64_POS execute sous qemu-riscv64 sur 20 000 couples
     (dividende 128 bits aleatoire ou aux bornes, diviseur > 0) contre l'arithmetique Python (cf. journal).

----------------------------------------------------------------------------------------------------------------------
MODIFICATION 1 : SDIV128_64_POS (nouvelle macro) + CVTIX + CVTXI
----------------------------------------------------------------------------------------------------------------------
ANCRE :
;	C O N V E	R S I O N	S   E N T	I E R  <->  F I X E D
;-----------------------------------------------------------------------------------------------------------------------

SUPPRIMER :
;			-----
macro			CVTIX					; Conversion de l'entier I en FIXED X de SMALL = NUMER/DENOM
;			-----					; Sont empiles I puis DENOM puis NUMER (I.D.N)
  POP_RBX								; a1 = NUMER : diviseur
  POP_RCX								; a2 = DENOM : multiplicateur
  POP_RAX								; a0 = I : multiplicande
  dd 0x02C50533							; mul a0, a0, a2		produit 64 bits bas
  dd 0x02B54533							; div a0, a0, a1		quotient signe
  PUSH_RAX							; Resultat FIXED X sur 64 bits
end macro
;			-----
macro			CVTXI					; Conversion du FIXED X de SMALL = NUMER/DENOM en INTEGER I
;			-----					; Sont empiles X puis NUMER puis DENOM (X.N.D)
  POP_RBX								; a1 = DENOM : diviseur
  POP_RCX								; a2 = NUMER : multiplicateur
  POP_RAX								; a0 = X : multiplicande
  dd 0x02C502B3							; mul  t0, a0, a2		produit 64 bits bas
  dd 0x02B2C533							; div  a0, t0, a1		quotient tronque
  dd 0x02B2E6B3							; rem  a3, t0, a1		reste signe

  dd 0x43F6D613							; srai a2, a3, 63		masque signe du reste
  dd 0x00C6C6B3							; xor  a3, a3, a2
  dd 0x40C686B3							; sub  a3, a3, a2		abs(reste)

  dd 0x0015F313							; andi t1, a1, 1		bit impair du denominateur
  dd 0x0015D593							; srli a1, a1, 1
  dd 0x006585B3							; add  a1, a1, t1		ceil(DENOM/2)

  dd 0x00B6E663							; bltu a3, a1, +12		si abs(reste) < demi, pas d'arrondi
  dd 0x00166613							; ori  a2, a2, 1		+1 si reste positif, -1 si negatif
  dd 0x00C50533							; add  a0, a0, a2		arrondi loin de zero au demi
  PUSH_RAX							; Resultat INTEGER I sur 64 bits
end macro
REMPLACER PAR :

;			--------------
macro			SDIV128_64_POS				; Division signee 128/64 : (a3:a0) / a1, avec a1 > 0
;			--------------				; Sortie : a0 = quotient signe, a3 = reste signe ; a1 preserve
;	Homologue de la macro arm64 du meme nom (x86 : imul/idiv sur RDX:RAX).
;	Chemin rapide si a3:a0 est deja l'extension de signe d'un entier 64 bits.
;	Sinon division non signee des valeurs absolues sur 64 iterations, puis restauration du signe.
;	Clobbers : a2, a4, t0, t1, t2. Branchements internes en B-type (RV_BRANCH), tous locaux.
  local slow, boucle, soustraire, suite, fin

; Produit deja representable sur 64 bits signes ?
  dd 0x43F55293							; srai t0, a0, 63
  RV_BRANCH	1, 13, 5, slow					; bne  a3, t0, slow

; Chemin rapide : division materielle 64/64, reste par rem.
  dd 0x02B54633							; div  a2, a0, a1
  dd 0x02B566B3							; rem  a3, a0, a1
  dd 0x00060513							; mv   a0, a2
  RV_BRANCH	0, 0, 0, fin					; beq  x0, x0, fin

slow:
; Masque de signe du dividende 128 (a4, conserve), puis valeur absolue 128 bits :
; (a3:a0) := (a3:a0) xor (a4:a4) - (a4:a4), l'emprunt de la soustraction basse passe par t0.
  dd 0x43F6D713							; srai a4, a3, 63
  dd 0x00E54533							; xor  a0, a0, a4
  dd 0x00E6C6B3							; xor  a3, a3, a4
  dd 0x00E532B3							; sltu t0, a0, a4		emprunt de (a0 - a4)
  dd 0x40E50533							; sub  a0, a0, a4
  dd 0x40E686B3							; sub  a3, a3, a4
  dd 0x405686B3							; sub  a3, a3, t0

; Division non signee 128/64.
; a3 = reste courant, a2 = bits bas du dividende, a0 = quotient, t1 = compteur.
  dd 0x00050613							; mv   a2, a0
  dd 0x00000513							; mv   a0, x0
  dd 0x04000313							; addi t1, x0, 64

boucle:
  dd 0x03F6D293							; srli t0, a3, 63		bit sortant de a3 (129e bit du reste decale)
  dd 0x03F65393							; srli t2, a2, 63
  dd 0x00169693							; slli a3, a3, 1
  dd 0x0076E6B3							; or   a3, a3, t2
  dd 0x00161613							; slli a2, a2, 1
  dd 0x00151513							; slli a0, a0, 1
  RV_BRANCH	1, 5, 0, soustraire				; bne  t0, x0, soustraire	(b.cs de l'arm64)
  RV_BRANCH	6, 13, 11, suite					; bltu a3, a1, suite	(b.lo de l'arm64)

soustraire:
  dd 0x40B686B3							; sub  a3, a3, a1
  dd 0x00156513							; ori  a0, a0, 1

suite:
  dd 0xFFF30313							; addi t1, t1, -1
  RV_BRANCH	1, 6, 0, boucle					; bne  t1, x0, boucle

; Restaurer le signe du quotient et du reste.
  dd 0x00E54533							; xor  a0, a0, a4
  dd 0x40E50533							; sub  a0, a0, a4
  dd 0x00E6C6B3							; xor  a3, a3, a4
  dd 0x40E686B3							; sub  a3, a3, a4

fin:
end macro
;			-----
macro			CVTIX					; Conversion de l'entier I en FIXED X de SMALL = NUMER/DENOM
;			-----					; Sont empiles I puis DENOM puis NUMER (I.D.N)
  POP_RBX								; a1 = NUMER : diviseur
  POP_RCX								; a2 = DENOM : multiplicateur
  POP_RAX								; a0 = I : multiplicande
  dd 0x02C516B3							; mulh a3, a0, a2		produit 128 bits : haut
  dd 0x02C50533							; mul  a0, a0, a2		produit 128 bits : bas
  SDIV128_64_POS
  PUSH_RAX							; Resultat FIXED X sur 64 bits
end macro
;			-----
macro			CVTXI					; Conversion du FIXED X de SMALL = NUMER/DENOM en INTEGER I
;			-----					; Sont empiles X puis NUMER puis DENOM (X.N.D)
  POP_RBX								; a1 = DENOM : diviseur
  POP_RCX								; a2 = NUMER : multiplicateur
  POP_RAX								; a0 = X : multiplicande
  dd 0x02C516B3							; mulh a3, a0, a2		produit 128 bits : haut
  dd 0x02C50533							; mul  a0, a0, a2		produit 128 bits : bas
  SDIV128_64_POS							; a0 = quotient tronque, a3 = reste signe

  dd 0x43F6D713							; srai a4, a3, 63		masque signe du reste
  dd 0x00E6C6B3							; xor  a3, a3, a4
  dd 0x40E686B3							; sub  a3, a3, a4		abs(reste)

  dd 0x0015F313							; andi t1, a1, 1		bit impair du denominateur
  dd 0x0015D593							; srli a1, a1, 1
  dd 0x006585B3							; add  a1, a1, t1		ceil(DENOM/2)  (shr + adc de la reference x86)

  dd 0x00B6E663							; bltu a3, a1, +12		si abs(reste) < demi, pas d'arrondi
  dd 0x00176713							; ori  a4, a4, 1		+1 si reste positif, -1 si negatif
  dd 0x00E50533							; add  a0, a0, a4		arrondi loin de zero au demi
  PUSH_RAX							; Resultat INTEGER I sur 64 bits
end macro

  end virtual
REMPLACER PAR :
  local bits

  end virtual

MODIFICATION 2 : LVA : (disp) parenthese
----------------------------------------------------------------------------------------------------------------------
ANCRE :
;	ou via QUAD_CONST + add pour les grandes valeurs.
SUPPRIMER :
  BASE_IN_RAX	lvl
  if disp = 0
;	a0 deja a la bonne valeur
  else if disp >= -2048  &  disp <= 2047
;	addi a0, a0, disp
    dd 0x00050513 or ((disp and 0xFFF) shl 20)
  else
    QUAD_CONST	5, disp						; t0 = disp
    dd 0x00550533							; add a0, a0, t0
  end if

  BASE_IN_RAX	lvl

;	a0 deja a la bonne valeur

  else
