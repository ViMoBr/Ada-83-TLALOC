CORRECTIF v3 — CLAUSES 'SIZE DES SCALAIRES : TABLE COTE EXPANDER + PRE-PASSE
(remplace le correctif "lecture gardee de SM_SIZE", desormais caduc)

Faits etablis par le temoin v2 + diana_NODES :
- La classe DIANA des noeuds Integer et enumeration n'a PAS d'attribut SM_SIZE
  (ni SM_REPRESENTATION pour enumeration). La clause 'SIZE d'un scalaire est un
  ITEM de declaration separe, DN_LENGTH_ENUM_REP, place APRES le type decl,
  aujourd'hui dispatche vers null (declarations.adb ~1466).
- La fonction D ne leve pas d'exception sur attribut absent : message + retour.
  L'idiome "exception when others" est donc inoperant (PIEGES : l'idiome de
  REPRESENTED_RECORD_SIZE_BITS est suspect du meme mal).
Motif reutilise : la table SET_GENERIC_ACTUAL_TYPE / GENERIC_ACTUAL_TYPE_OF de
UTILS — etat cote expander quand DIANA n'a pas de slot.

  SIZE_CLAUSE_MAX	:constant		:= 64;					--| clauses rares : table fixe, recherche lineaire
  type SIZE_CLAUSE_ENTRY	is record
			     SPEC	: TREE;
			     BITS	: INTEGER;
			   end record;
  SIZE_CLAUSE_TBL	: array ( 1 .. SIZE_CLAUSE_MAX ) of SIZE_CLAUSE_ENTRY;
  SIZE_CLAUSE_CNT	: NATURAL	:= 0;

			--==============--
  procedure		  SET_SIZE_CLAUSE	( TYPE_SPEC :TREE; BITS :INTEGER )
  is			--==============--
  begin
    for  I in 1 .. SIZE_CLAUSE_CNT  loop						-- re-pose (recompilation d'unite) : ecraser
      if  SIZE_CLAUSE_TBL( I ).SPEC = TYPE_SPEC  then
        SIZE_CLAUSE_TBL( I ).BITS := BITS;
        return;
      end if;
    end loop;
    if  SIZE_CLAUSE_CNT >= SIZE_CLAUSE_MAX  then
      PUT_LINE( "; EXPANDER.UTILS.SET_SIZE_CLAUSE : table pleine !" );
      raise PROGRAM_ERROR;
    end if;
    SIZE_CLAUSE_CNT := SIZE_CLAUSE_CNT + 1;
    SIZE_CLAUSE_TBL( SIZE_CLAUSE_CNT ) := ( SPEC => TYPE_SPEC, BITS => BITS );
  end	  SET_SIZE_CLAUSE;
	--==============--

			--===============--
  function		  SIZE_CLAUSE_BITS	( TYPE_SPEC :TREE )	return INTEGER
  is			--===============--
  begin
    for  I in 1 .. SIZE_CLAUSE_CNT  loop
      if  SIZE_CLAUSE_TBL( I ).SPEC = TYPE_SPEC  then
        return SIZE_CLAUSE_TBL( I ).BITS;
      end if;
    end loop;
    return 0;
  end	  SIZE_CLAUSE_BITS;
	--===============--

			--====================--
  procedure		  PRE_SCAN_SIZE_CLAUSES	( DECL_S :TREE )
  is			--====================--
  -- Releve les "for T'SIZE use N" d'une liste de declarations AVANT sa
  -- compilation : la clause est un item DN_LENGTH_ENUM_REP place apres le
  -- type decl (RM83 13.1 : dans la meme partie declarative), et la classe
  -- DIANA des specs scalaires n'a pas de slot pour la recevoir.
    SCAN_SEQ	: SEQ_TYPE	:= LIST( DECL_S );
    ITEM	: TREE;
  begin
    while  not IS_EMPTY( SCAN_SEQ )  loop
      POP( SCAN_SEQ, ITEM );

      if  ITEM.TY = DN_LENGTH_ENUM_REP  then
        declare
	ATTR	: TREE	:= D( AS_NAME, ITEM );
        begin
	if  ATTR.TY = DN_ATTRIBUTE
	and then  PRINT_NAME( D( LX_SYMREP, D( AS_USED_NAME_ID, ATTR ) ) ) = "SIZE"
	then
	  declare
	    TYPE_ID	: TREE	:= D( SM_DEFN, D( AS_NAME, ATTR ) );
	    VAL	: TREE	:= D( AS_EXP, ITEM );
	    SPEC	: TREE;
	  begin
	    while  VAL.TY = DN_PARENTHESIZED  or else  VAL.TY = DN_CONVERSION  loop
	      VAL := D( AS_EXP, VAL );
	    end loop;

	    if  TYPE_ID.TY = DN_TYPE_ID  and then  VAL.TY = DN_NUMERIC_LITERAL  then
	      SPEC := FULL_VIEW( D( SM_TYPE_SPEC, TYPE_ID ) );
	      SET_SIZE_CLAUSE( SPEC, DI( SM_VALUE, VAL ) );
	      if  SPEC.TY = DN_INTEGER  or else  SPEC.TY = DN_ENUMERATION  then
	        SET_SIZE_CLAUSE( D( SM_BASE_TYPE, SPEC ), DI( SM_VALUE, VAL ) );
	      end if;
	    end if;
	  end;
	end if;
        end;
      end if;
    end loop;
  end	  PRE_SCAN_SIZE_CLAUSES;
	--====================--
