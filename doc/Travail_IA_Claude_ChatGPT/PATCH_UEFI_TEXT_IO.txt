PATCH codi_x86_64U.finc  --  palier 2 : TEXT_IO / PUT_LINE sous UEFI
Base : votre version du 16/09, celle que vous m'avez transmise.
Verifie : DIS_BONJOUR assemble en 34816 octets et affiche " Bonjour " sous OVMF.

Six retouches, dans l'ordre du fichier.
  1. tampon de conversion et memoire du dernier octet, en .bss
  2. constante de taille du tampon
  3. la couche stdout : WIN64_WRITE_STDOUT -> UEFI_WRITE_STDOUT
  4. neutralisation deterministe des services non encore portes
  5. retour de "purge data?"  (TEXT_IO declare un STATOFS nomme DATA)
  6. .reloc emis a la main, consequence du point 5

========================================================================================================
PATCH 1  --  .bss : tampon de conversion UTF-16 et memoire du dernier octet
========================================================================================================
--- ANCRE (ne pas modifier, sert a se reperer) --------------------------------------------------------
UEFI_IMAGE_HANDLE:	rq 1
UEFI_SYSTEM_TABLE:	rq 1
--- BLOC A INSERER -----------------------------------------------------------------------------------
UEFI_TXT_BUF:		rw UEFI_TXT_MAX + 4			; tampon de conversion UTF-16 (voir UEFI_WRITE_STDOUT)
UEFI_TXT_LAST:		rb 1					; dernier octet emis (pour ne pas doubler le CR)
(a inserer juste apres l'ancre, avant la ligne blanche qui precede UEFI_HEAP_BASE)

PATCH 2  --  taille du tampon, a cote des offsets UEFI

--- BLOC A INSERER -----------------------------------------------------------------------------------
UEFI_TXT_MAX		= 256			; CHAR16 par tournee de UEFI_WRITE_STDOUT

========================================================================================================
PATCH 3  --  la couche stdout
========================================================================================================
Le contrat d'entree (RDX = tampon d'octets, R8D = longueur) est conserve tel quel,
donc SYS_PUT_CHAR et SYS_PUT_STR ne changent que d'appelee. Tout le bloc ci-dessous,
du commentaire d'en-tete jusqu'au "end macro" de SYS_PUT_STR, est remplace.
--- BLOC A SUPPRIMER ---------------------------------------------------------------------------------
; Windows x64 : WriteFile( STDOUT, buffer, length, &written, NULL ).
; A l'entree : RDX = buffer, R8D = length.
macro            WIN64_WRITE_STDOUT
  db 0x48, 0x89, 0xE0						; mov rax,rsp
  db 0x48, 0x83, 0xE4, 0xF0						; and rsp,-16
  db 0x48, 0x83, 0xEC, 0x40						; sub rsp,64
  db 0x48, 0x89, 0x44, 0x24, 0x30					; mov [rsp+48],rax : ancien RSP
  db 0x48, 0xC7, 0x44, 0x24, 0x20, 0, 0, 0, 0				; [rsp+32] = lpOverlapped = NULL
  db 0x4C, 0x8D, 0x4C, 0x24, 0x28					; lea r9,[rsp+40] : &written
  db 0x48, 0x8B, 0x0D						; mov rcx,[rip+WIN_STDOUT_HANDLE]
;  dd WIN_STDOUT_HANDLE - ($ + 4)
  db 0xFF, 0x15							; call qword [rip+WriteFileW64]
;  dd WriteFileW64 - ($ + 4)
  db 0x48, 0x8B, 0x64, 0x24, 0x30					; mov rsp,[rsp+48]
end macro

;                       ------------
macro                   SYS_PUT_CHAR					; stdout Windows
;                       ------------
  db 0x48, 0x89, 0xEA						; mov rdx,rbp : @caractere LLIR
  db 0x41, 0xB8, 1, 0, 0, 0						; mov r8d,1
  WIN64_WRITE_STDOUT
  DROP
end macro

;                       -----------
macro                   SYS_PUT_STR					; stdout Windows
;                       -----------
  POP_RSI								; RSI = @(@DATA | @USEINFO)
  db 0x48, 0x8B, 0x46, 0x08						; mov rax,[rsi+8] : @USEINFO
  db 0x44, 0x8B, 0x40, 0x0C						; mov r8d,[rax+12] : last
  db 0x41, 0xFF, 0xC0						; inc r8d
  db 0x44, 0x2B, 0x40, 0x08						; sub r8d,[rax+8] : length
  db 0x48, 0x8B, 0x16						; mov rdx,[rsi] : @DATA
  WIN64_WRITE_STDOUT
end macro
--- BLOC A INSERER -----------------------------------------------------------------------------------
; UEFI : ConOut->OutputString( ConOut, CHAR16* ). Trois ecarts avec WriteFile :
;   - le firmware ne connait que l'UTF-16LE : il faut promouvoir chaque octet ;
;   - il ne connait qu'une chaine terminee par un CHAR16 nul, pas un couple
;     (adresse, longueur) : il faut recopier dans un tampon ;
;   - il ne traduit pas LF en CRLF : on le fait ici, au plus bas niveau, pour
;     que TEXT_IO reste rigoureusement identique sur les quatre cibles.
;
; A l'entree : RDX = tampon d'octets, R8D = longueur.  Meme contrat que
; WIN64_WRITE_STDOUT, donc SYS_PUT_CHAR et SYS_PUT_STR sont inchanges.
; L'etat de boucle est tenu dans RSI, RDI et RBX, non volatils dans l'ABI
; Microsoft x64 : ils survivent a l'appel firmware, qui detruit RAX, RCX,
; RDX et R8..R11.
;			------------------
macro			UEFI_WRITE_STDOUT
;			------------------
  local w_flush, w_fill, w_store, w_emit, w_done
  db 0x48, 0x89, 0xD6						; mov rsi,rdx		source
  db 0x44, 0x89, 0xC3						; mov ebx,r8d		compte restant (zero-etendu)
  db 0x48, 0x85, 0xDB						; test rbx,rbx
  db 0x74, w_done - ($ + 1)					; jz w_done
w_flush:
  db 0x48, 0x8D, 0x3D						; lea rdi,[rip+UEFI_TXT_BUF]
  dd UEFI_TXT_BUF - ($ + 4)
  db 0xB9							; mov ecx,UEFI_TXT_MAX	places libres
  dd UEFI_TXT_MAX
w_fill:
  db 0x48, 0x85, 0xDB						; test rbx,rbx
  db 0x74, w_emit - ($ + 1)					; jz w_emit
  db 0x83, 0xF9, 0x02						; cmp ecx,2		(reserver la place d'un CRLF)
  db 0x7C, w_emit - ($ + 1)					; jl w_emit
  db 0x0F, 0xB6, 0x06						; movzx eax,byte [rsi]
  db 0x48, 0xFF, 0xC6						; inc rsi
  db 0x48, 0xFF, 0xCB						; dec rbx
  db 0x83, 0xF8, 0x0A						; cmp eax,10		LF ?
  db 0x75, w_store - ($ + 1)					; jne w_store
			; LF : inserer un CR, SAUF si le caractere precedent etait deja un CR.
			; TEXT_IO.NEW_LINE emet explicitement CR puis LF, comme sur les trois
			; autres cibles, mais par DEUX appels SYS_PUT_CHAR distincts : le
			; souvenir doit donc survivre a l'appel, d'ou l'octet en memoire.
			; Un LF nu (SYS_PUT_STR direct du chemin exception) reste ainsi traduit.
  db 0x80, 0x3D							; cmp byte [rip+UEFI_TXT_LAST],13
  dd UEFI_TXT_LAST - ($ + 5)
  db 0x0D
  db 0x74, w_store - ($ + 1)					; je w_store		CR deja emis
  db 0x66, 0xC7, 0x07, 0x0D, 0x00				; mov word [rdi],13
  db 0x48, 0x83, 0xC7, 0x02					; add rdi,2
  db 0xFF, 0xC9							; dec ecx
w_store:
  db 0x88, 0x05							; mov [rip+UEFI_TXT_LAST],al	memoriser
  dd UEFI_TXT_LAST - ($ + 4)
  db 0x66, 0x89, 0x07						; mov [rdi],ax		octet promu en CHAR16
  db 0x48, 0x83, 0xC7, 0x02					; add rdi,2
  db 0xFF, 0xC9							; dec ecx
  db 0xEB, w_fill - ($ + 1)					; jmp w_fill
w_emit:
  db 0x66, 0xC7, 0x07, 0x00, 0x00				; mov word [rdi],0	terminateur CHAR16
  db 0x4C, 0x8B, 0x15						; mov r10,[rip+UEFI_SYSTEM_TABLE]
  dd UEFI_SYSTEM_TABLE - ($ + 4)
  db 0x49, 0x8B, 0x4A, ConOut					; mov rcx,[r10+0x40]	RCX = ConOut
  db 0x4C, 0x8B, 0x51, OutputString				; mov r10,[rcx+0x08]	R10 = &OutputString
  db 0x48, 0x8D, 0x15						; lea rdx,[rip+UEFI_TXT_BUF]
  dd UEFI_TXT_BUF - ($ + 4)
  db 0x48, 0x89, 0xE0						; mov rax,rsp
  db 0x48, 0x83, 0xE4, 0xF0					; and rsp,-16
  db 0x48, 0x83, 0xEC, 0x30					; sub rsp,48
  db 0x48, 0x89, 0x44, 0x24, 0x28				; mov [rsp+40],rax
  db 0x41, 0xFF, 0xD2						; call r10
  db 0x48, 0x8B, 0x64, 0x24, 0x28				; mov rsp,[rsp+40]
  db 0x48, 0x85, 0xDB						; test rbx,rbx
  db 0x75, w_flush - ($ + 1)					; jnz w_flush		il reste des octets
w_done:
end macro

;                       ------------
macro                   SYS_PUT_CHAR					; stdout UEFI
;                       ------------
  db 0x48, 0x89, 0xEA						; mov rdx,rbp : @caractere LLIR
  db 0x41, 0xB8, 1, 0, 0, 0					; mov r8d,1
  UEFI_WRITE_STDOUT
  DROP
end macro

;                       -----------
macro                   SYS_PUT_STR					; stdout UEFI
;                       -----------
  POP_RSI								; RSI = @(@DATA | @USEINFO)
  db 0x48, 0x8B, 0x46, 0x08					; mov rax,[rsi+8] : @USEINFO
  db 0x44, 0x8B, 0x40, 0x0C					; mov r8d,[rax+12] : last
  db 0x41, 0xFF, 0xC0						; inc r8d
  db 0x44, 0x2B, 0x40, 0x08					; sub r8d,[rax+8] : length
  db 0x48, 0x8B, 0x16						; mov rdx,[rsi] : @DATA
  UEFI_WRITE_STDOUT
end macro

========================================================================================================
PATCH 4  --  neutralisation des services non encore portes
========================================================================================================
Vous aviez commente les lignes "dd XxxW64 - ($ + 4)" pour debloquer l'assemblage.
Le probleme est que le "db 0xFF, 0x15" qui precede reste, lui, dans le flux : il
avale les 4 octets suivants comme deplacement et desynchronise la suite de la macro.
TEXT_IO expanse reellement SYS_FILE_WRITE (trois fois), donc ces octets sont bien
emis dans DIS_BONJOUR.EFI. Ils ne sont pas executes ici, mais ils le seront un jour.

La regle : supprimer la PAIRE (instruction + ligne commentee) et la remplacer par
une instruction qui force la branche d'echec deja presente dans la macro. Quatre
formes selon le symbole commente. 22 sites au total.

  (a) ";  dd CreateFileAW64 - ($ + 4)"                 -> resultat = -1 attendu par l'appelant
      SUPPRIMER :
  db 0xFF, 0x15							; call [rip+CreateFileAW64]
;  dd CreateFileAW64 - ($ + 4)
      INSERER :
  db 0x48, 0xC7, 0xC0, 0xFF, 0xFF, 0xFF, 0xFF			; mov rax,-1 : service non porte sur UEFI

  (b) ";  dd GetSystemTimeAsFileTimeW64 - ($ + 4)"     -> RCX pointe le tampon, on le met a zero
      SUPPRIMER :
  db 0xFF, 0x15							; call [rip+GetSystemTimeAsFileTimeW64]
;  dd GetSystemTimeAsFileTimeW64 - ($ + 4)
      INSERER :
  db 0x48, 0xC7, 0x01, 0x00, 0x00, 0x00, 0x00			; mov qword [rcx],0 : service non porte sur UEFI

  (c) ";  dd WIN_STDIN_HANDLE - ($ + 4)"               -> il n'y a pas de handle sous UEFI
      SUPPRIMER :
  db 0x48, 0x8B, 0x0D                                ; mov rcx,[rip+WIN_STDIN_HANDLE]
;  dd WIN_STDIN_HANDLE - ($ + 4)
      INSERER :
  db 0x48, 0x31, 0xC9						; xor rcx,rcx : handle absent sous UEFI

  (d) tous les autres (GetConsoleModeW64, SetConsoleModeW64, ReadFileW64,
      WriteFileW64, SetFilePointerExW64, GetFileSizeExW64, CloseHandleW64,
      DeleteFileAW64)  -> la macro teste deja EAX et branche sur son etiquette d'echec
      SUPPRIMER :
  db 0xFF, 0x15							; call [rip+XxxW64]
;  dd XxxW64 - ($ + 4)
      INSERER :
  db 0x31, 0xC0							; xor eax,eax : service non porte -> branche d'echec

Apres quoi "objdump -d -j .text BOOTX64.EFI | grep -c 'ff 15'" doit renvoyer 0.
Les services concernes rendent alors proprement -1, ce que TEXT_IO traduit en
DEVICE_ERROR ou USE_ERROR au lieu de partir dans le decor.

========================================================================================================
PATCH 5  --  retablir purge data?
========================================================================================================
TEXT_IO.FINC contient "STATOFS DATA, 16384, 1", et STATOFS fait "DATA = $".
Or "data" est une macro de format/pe.inc, insensible a la casse : elle intercepte
la ligne et l'assemblage echoue sur "invalid expression". C'est precisement la
raison du "purge data?" de codi_x86_64W.finc. Il faut le remettre.
--- ANCRE (ne pas modifier, sert a se reperer) --------------------------------------------------------
section '.text' code readable executable
--- BLOC A INSERER -----------------------------------------------------------------------------------

purge data?

========================================================================================================
PATCH 6  --  .reloc emis a la main (consequence du patch 5)
========================================================================================================
Le "purge data?" prive le postpone de la macro "data", dont la forme
"section '.reloc' fixups data discardable" a besoin. Et fasmg ne sait pas defaire
un purge : "restore data?" ne remet pas une macro. On fait donc a la main ce que
"data fixups" faisait : appeler PE.fixups, puis renseigner soi-meme l'entree 5 du
repertoire des donnees (adresse et taille de la table de relocations).
--- ANCRE (ne pas modifier, sert a se reperer) --------------------------------------------------------
;	La section .reloc doit etre emise APRES tout ce qui engendre des relocations
;	(le FAS TLALOC, qui suit l'include) : on la differe par postpone.
--- BLOC A SUPPRIMER ---------------------------------------------------------------------------------
postpone
	section '.reloc' fixups data discardable
end postpone
--- BLOC A INSERER -----------------------------------------------------------------------------------
postpone
	section '.reloc' data discardable readable
  namespace PE
	RELOC_RVA = $ - IMAGE_BASE
  end namespace
	PE.fixups
  namespace PE
	store RELOC_RVA : 4 at PE:RvaAndSizes.Rva  + 5*RvaAndSizes.ENTRY_LENGTH
	store $ - IMAGE_BASE - RELOC_RVA : 4 at PE:RvaAndSizes.Size + 5*RvaAndSizes.ENTRY_LENGTH
  end namespace
end postpone
