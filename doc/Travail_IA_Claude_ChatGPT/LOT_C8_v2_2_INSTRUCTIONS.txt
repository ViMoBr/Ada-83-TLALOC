CE QUE LES SITES univ_ops REVELENT
----------------------------------
    function U_INT ( V :VECTOR ) return TREE is
      VDP : VECTOR_PAIRS;  for VDP use at V'ADDRESS;
La cible est un PARAMETRE in (SM_DEFN de V = DN_IN_ID) -- rejete par
OVERLAY_TARGET (hors CLASS_VC_NAME), d'ou le TROU. C'est une TROISIEME
sorte de slot : celui d'un parametre COMPOSITE porte l'@DOUBLET de
l'actuel (protocole n 91/94, celui-la meme que C3 a re-confirme).
- Pas d'equation possible : le symbole V_disp n'existe pas, seul
  -V_ofs existe (PRMzone), et son contenu est un @doublet, pas un
  data_ptr.
- Le bon geste, deux indirections a l'elaboration :
      La  lvl, -V_ofs        ; slot du parametre = @doublet de l'actuel
      La  ,  0               ; data_ptr de l'actuel
      Sa  lvl, VDP_disp
  __u de VDP inchange (descripteur _VECTOR_PAIRS : la vue de
  reinterpretation -- L,S communs, P par-dessus D, pragma PACK compris :
  le layout est l'affaire des namespaces de types, pas de l'overlay).

Grille des sortes apres v2.2 (cible en colonnes) :
                 scalaire     composite      PARAM composite   autre
  scalaire       EQUATION     TROU           TROU (E2)         TROU
  array          LVA @slot    EQUATION       TROU (E3)         TROU
  record         TROU         EQUATION       La/-ofs+La,0 (E4) TROU
E1 ouvre OVERLAY_TARGET aux DN_IN_ID ; E2/E3 EXCLUENT explicitement les
parametres des chemins equation/LVA (le symbole _disp n'existerait pas
-- fasmg echouerait bruyamment, mais le TROU discrimine est plus juste).
Array-sur-parametre : non observe, TROU (carnet N9).

Conventions : tabulations REELLES ; SUPPRIMER = bloc exact, sa 1re
ligne sert d'ancre quand elle est REPRISE en tete du remplacement.

--- E1 --- helper OVERLAY_TARGET : accepter la cible parametre in

SUPPRIMER (bloc unique) :
	TARGET := D( SM_DEFN, PREFIX );
	if  TARGET.TY not in CLASS_VC_NAME
	or else  DB( SM_RENAMES_OBJ, TARGET )
	or else  DI( CD_LEVEL, TARGET ) /= INTEGER( CODI.CUR_LEVEL )
	then
	  return TREE_VOID;
	end if;

REMPLACER PAR :
	TARGET := D( SM_DEFN, PREFIX );
	if  TARGET.TY in CLASS_VC_NAME  then
	  if  DB( SM_RENAMES_OBJ, TARGET )  then							-- le _disp d'un renames porte une ADRESSE
	    return TREE_VOID;
	  end if;
	elsif  TARGET.TY = DN_IN_ID  then								-- C8 (univ_ops) : cible = PARAMETRE in --
	  null;											--| slot = VALEUR (scalaire) ou @doublet (composite, n 91/94)
	else
	  return TREE_VOID;
	end if;
	if  DI( CD_LEVEL, TARGET ) /= INTEGER( CODI.CUR_LEVEL )  then
	  return TREE_VOID;
	end if;

--- E2 --- COMPILE_VC_NAME_INTEGER : exclure les parametres de
    l'equation scalaire (garde de coherence)

SUPPRIMER (bloc unique, 3 lignes) :
	  if  TGT = TREE_VOID  or else  INIT_EXP /= TREE_VOID
	  or else  CODI.FULL_TYPE_VIEW( D( SM_OBJ_TYPE, TGT ) ).TY  not in  CLASS_SCALAR  then		-- slot cible = VALEUR : l'equation exige la meme sorte des deux cotes

REMPLACER PAR :
	  if  TGT = TREE_VOID  or else  INIT_EXP /= TREE_VOID
	  or else  TGT.TY  not in  CLASS_VC_NAME							-- parametre : pas de symbole _disp, pas d'equation
	  or else  CODI.FULL_TYPE_VIEW( D( SM_OBJ_TYPE, TGT ) ).TY  not in  CLASS_SCALAR  then		-- slot cible = VALEUR : l'equation exige la meme sorte des deux cotes

--- E3 --- COMPILE_ARRAY_VAR, calcul des sortes : exclure les
    parametres des chemins equation/LVA

SUPPRIMER (bloc unique, 3 lignes) :
	OVERLAY_EQUATE   := OVERLAY_TGT_TYPE.TY = DN_ARRAY   or else  OVERLAY_TGT_TYPE.TY = DN_CONSTRAINED_ARRAY
		     or else  OVERLAY_TGT_TYPE.TY = DN_RECORD  or else  OVERLAY_TGT_TYPE.TY = DN_CONSTRAINED_RECORD;
	OVERLAY_LVA      := OVERLAY_TGT_TYPE.TY in CLASS_SCALAR;

REMPLACER PAR :
	OVERLAY_EQUATE   := OVERLAY_TGT.TY in CLASS_VC_NAME						-- parametre : pas de symbole _disp (TROU, cf. grille v2.2)
		     and then  ( OVERLAY_TGT_TYPE.TY = DN_ARRAY   or else  OVERLAY_TGT_TYPE.TY = DN_CONSTRAINED_ARRAY
		     or else  OVERLAY_TGT_TYPE.TY = DN_RECORD  or else  OVERLAY_TGT_TYPE.TY = DN_CONSTRAINED_RECORD );
	OVERLAY_LVA      := OVERLAY_TGT.TY in CLASS_VC_NAME  and then  OVERLAY_TGT_TYPE.TY in CLASS_SCALAR;

--- E4 --- COMPILE_RECORD_VAR, emission : la branche parametre

SUPPRIMER (bloc unique, 4 lignes -- le  if DO_OVERLAY  et l'equation ;
les lignes  VAR __u  qui suivent restent en place, DANS la branche) :
	if  DO_OVERLAY  then
	  PUT( VC_STR & "_disp = " & PRINT_NAME( D( LX_SYMREP, OVERLAY_TGT ) ) & "_disp" );		-- C8 : OVERLAY, equation fasmg -- meme cellule data_ptr, pas de __dat
	  if  CODI.DEBUG  then  PUT( tab50 & "; variable record OVERLAY : data_ptr partage" ); end if;
	  NEW_LINE;

REMPLACER PAR :
	if  DO_OVERLAY  then
	  if  OVERLAY_TGT.TY = DN_IN_ID  then								-- C8 : cible = PARAMETRE composite (univ_ops) --
	    PUT( "VAR " & VC_STR & "_disp, q" );							--| son slot porte l'@doublet de l'ACTUEL (n 91/94) :
	    if  CODI.DEBUG  then  PUT( tab50 & "; variable record OVERLAY sur parametre" ); end if;	--| data_ptr := [[-ofs]+0] -- pas d'equation possible
	    NEW_LINE;
	    PUT_LINE( tab & "La  " & LVL_STR & ", -" & PRINT_NAME( D( LX_SYMREP, OVERLAY_TGT ) ) & "_ofs" );
	    PUT_LINE( tab & "La  ,  0" );
	    PUT( tab & "Sa" & tab & LVL_STR & ", " & VC_STR & "_disp" );
	    if  CODI.DEBUG  then  PUT( tab50 & "; record data ptr at _disp (OVERLAY, data de l'actuel)" ); end if;
	    NEW_LINE;
	  else
	    PUT( VC_STR & "_disp = " & PRINT_NAME( D( LX_SYMREP, OVERLAY_TGT ) ) & "_disp" );		-- C8 : OVERLAY, equation fasmg -- meme cellule data_ptr, pas de __dat
	    if  CODI.DEBUG  then  PUT( tab50 & "; variable record OVERLAY : data_ptr partage" ); end if;
	    NEW_LINE;
	  end if;

     -- un overlay ecrirait la copie, pas l'actuel (semantique
     differente de l'idiome). TROU via E2 ; a n'ouvrir que sur temoin
     avec cette semantique explicitement voulue.

     LRM 13.5 (erroneous a la charge du programmeur), et c'est
     precisement ce que le passage par reference rend observable.
     univ_ops ne fait a priori que LIRE ; le temoin fige neanmoins la
     semantique par-reference -- si un jour les petits composites
     passaient par copie, 7.2 le detecterait.
