-- expander-instructions.adb, INVERSE_RECURSE_ON_PARAMETERS,
-- branche DN_INDEXED (l. 830-835) -- VERSION REVISEE, calquee sur le
-- precedent SELARG de la branche DN_SELECTED (l. 606-648), qui normalise
-- deja les actuals composites en @doublet.  Remplace le patch precedent
-- (helper WRAP_COMPOSITE_ACTUAL_DOUBLET) : meme motif, meme style, meme
-- source de type (SM_EXP_TYPE de l'ACTUAL, deprivatise), pour qu'un seul
-- idiome existe dans le fichier.

        elsif  ACT_PRM.TY = DN_INDEXED  then					-- COMPOSANT INDEXE EN ACTUAL

	declare
	  ACT_TYPE	: TREE	:= D( SM_EXP_TYPE, ACT_PRM );
	begin
	  while  ACT_TYPE.TY = DN_PRIVATE  or else  ACT_TYPE.TY = DN_L_PRIVATE  loop
	    ACT_TYPE := D( SM_TYPE_SPEC, ACT_TYPE );
	  end loop;

	  if  ACT_TYPE.TY = DN_ARRAY  or else  ACT_TYPE.TY = DN_CONSTRAINED_ARRAY  or else  ACT_TYPE.TY = DN_RECORD
	  then
			-- Composite : meme normalisation @doublet que SELARG.
			-- CODE_EXP laisse l'@data de l'element ; on l'enveloppe
			-- dans [ @element | use__info du type ] et on passe
			-- l'adresse du doublet (segfault TO_INT( TTR(I) ) sinon :
			-- le callee fait La ,0 et dereference la VALEUR).
	    declare
	      ANON	:constant STRING	:= "INDARG_" & NEW_LABEL;
	      LVL_STR	:constant STRING	:= IMAGE( CODI.CUR_LEVEL );
	      TYPE_NAME	: TREE		:= D( XD_SOURCE_NAME, ACT_TYPE );
	      TYPE_NAME_STR	:constant STRING	:= '_' & PRINT_NAME( D( LX_SYMREP, TYPE_NAME ) );
	    begin
	      PUT_LINE( tab & "VAR " & ANON & "_disp, q" );
	      PUT_LINE( tab & "VAR " & ANON & "__u, q" );

	      EXPRESSIONS.CODE_EXP( ACT_PRM );					-- @element (data nue)
	      PUT_LINE( tab & "Sa" & tab & LVL_STR & ", " & ANON & "_disp" );

	      PUT( tab & "La " & IMAGE( DI( CD_LEVEL, ACT_TYPE ) ) & ", " );
	      CODI.REGIONS_PATH( TYPE_NAME );
	      PUT_LINE( TYPE_NAME_STR & ".use__info" );
	      PUT_LINE( tab & "Sa" & tab & LVL_STR & ", " & ANON & "__u" );

	      PUT_LINE( tab & "LVA " & LVL_STR & ", " & ANON & "_disp" );
	    end;

	  else
	    if  FRM_PRM_ID.TY = DN_IN_ID  then
	      EXPRESSIONS.CODE_EXP( ACT_PRM );					-- in : rvalue scalaire
	    else
	      EXPRESSIONS.CODE_OBJECT_ADDRESS( ACT_PRM );			-- out / in out : adresse seule
	    end if;
	  end if;
	end;
